   197→/** Registro IDB → fila LIVIANA (sin Blob, sin acta — la foto duerme). */
   198→function filaDeRegistro(r: RegistroMasiva): FilaMasiva {
   199→  const acta = r.acta;
   200→  return {
   201→    id: r.id,
   202→    nombre: r.nombre,
   203→    agregadaEn: r.agregadaEn,
   204→    estado: r.estado,
   205→    statusActa: acta?.status,
   206→    score: acta?.score,
   207→    nivel: acta?.clasificacion?.nivel ?? null,
   208→    pagina: acta ? paginaDeActa(acta) : null,
   209→    motivo: acta?.rechazo?.tipo ?? null,
   210→    thumb: r.thumb,
   211→    // D72: el título compuesto usa el formato compacto cuando r.acta existe
   212→    infoTitulo: (r.acta ? tituloDeActa(r.acta) : r.infoTitulo) ?? null,
   213→    enviadaAuto: r.acta?.status === "ENVIADA" && !!r.acta?.enviadoAutomaticamente,
   214→    // KIT e14-masiva-ux-v1 (T3.2): el estado de envío se deriva DEL REGISTRO
   215→    // (kit-literal: hashSha256 === "EN COLA") — sobrevive la recarga.
   216→    envioEnCola: acta?.hashSha256 === "EN COLA",
   217→    // KIT e14-masiva-ux-v1 (T2): el motivo corto de las filas ERROR sin
   218→    // acta (guards de extracción/formato) sobrevive la recarga.
   219→    error:
   220→      r.estado === "ERROR"
   221→        ? (r.error ?? (r.acta ? null : "ERROR DE COLA"))
   222→        : null,
   223→    // KIT e14-masiva-tiempos-v1 (PR-1): dedup persistente — sobrevive la
   224→    // recarga (la fila sigue DUPLICADA y el runner puede copiar igual).
   225→    hashDedup: r.hashDedup ?? null,
   226→    dupOf: r.estado === "DUPLICADA" ? (r.dupOf ?? null) : null,
   227→    // D63-A2 (T1): viaPdf sobrevive la recarga (el override de origen del
   228→    // runner sigue funcionando a mitad de corrida). Ausente = no-PDF.
   229→    ...(r.viaPdf ? { viaPdf: true } : {}),
   230→    // e63k (F1): la serie impresa (mapa de identidad de la capa 2) y el
   231→    // override del dueño sobreviven la recarga. Aditivos al final del
   232→    // objeto retornado (contrato D57). Ausentes = registros previos.
   233→    serie: r.serie ?? null,
   234→    ...(r.forzarDedupOff ? { forzarDedupOff: true } : {}),
   235→    // e64-qr-v1 (T-0.2): el estado de la CAPA 0 QR sobrevive la recarga.
   236→    // Derivación honesta para registros e63m (traen qrPayload pero nacieron
   237→    // antes de qrEstado): el payload EXISTE ⇒ fue leído. undefined → la fila
   238→    // pinta sin chip ("n/d" — la capa 0 no corrió o el registro es previo).
   239→    qrEstado: r.qrEstado ?? (r.qrPayload ? "leido" : undefined),
   240→    // e65-dedup-enviadas-v1 (T-6): la YA-ENVIADA y sus llaves sobreviven la
   241→    // recarga — hashBytes SIEMPRE (llave de dedup viva para la cola viva
   242→    // tras recargar), enviadaDe SOLO cuando el estado lo es (SOLO-VISUAL).
   243→    hashBytes: r.hashBytes ?? null,
   244→    ...(r.estado === "YA-ENVIADA" ? { enviadaDe: r.enviadaDe ?? null } : {}),
   245→    // KIT e73-sondeo-qr-precoz-v1 (§E) — la telemetría del sondeo sobrevive
   246→    // la recarga (chip «Sondeo QR N/M» + p50/p95 del informe). Aditivos AL
   247→    // FINAL del objeto retornado (contrato D57). Ausentes = registros previos.
   248→    ...(r.sondeoQrMs != null ? { sondeoQrMs: r.sondeoQrMs } : {}),
   249→    ...(r.sondeoQrTier != null ? { sondeoQrTier: r.sondeoQrTier } : {}),
   250→  };
   251→}
