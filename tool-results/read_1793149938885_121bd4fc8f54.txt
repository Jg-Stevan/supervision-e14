   278→// KIT e14-ocr-escalonado-v1 (§2.1): cap PARAMETRIZABLE — default DET_CAP_PROD
   279→// (byte a byte el flujo de hoy); el OCR escalonado pasa DET_CAP_RAPIDO en la
   280→// 1.ª pasada.
   281→function preDet(
   282→  src: HTMLImageElement,
   283→  cap: number = DET_CAP_PROD,
   284→): { canvas: HTMLCanvasElement; ops: OpsDet } {
   285→  let w = src.naturalWidth || src.width;
   286→  let h = src.naturalHeight || src.height;
   287→  let ratioW = 1;
   288→  let ratioH = 1;
   289→
   290→  // op1 — reduce_max_side(img, 1920): int(h*ratio) luego round32 cada lado
   291→  if (Math.max(h, w) > 1920) {
   292→    const ratio = 1920 / Math.max(h, w);
   293→    const nh = round32(Math.floor(h * ratio));
   294→    const nw = round32(Math.floor(w * ratio));
   295→    ratioW *= nw / w;
   296→    ratioH *= nh / h;
   297→    w = nw;
   298→    h = nh;
   299→  }
   300→
   301→  // op2 — increase_min_side(img, 30): idem con el lado menor
   302→  if (Math.min(h, w) < 30) {
   303→    const ratio = 30 / Math.min(h, w);
   304→    const nh = round32(Math.floor(h * ratio));
   305→    const nw = round32(Math.floor(w * ratio));
   306→    ratioW *= nw / w;
   307→    ratioH *= nh / h;
   308→    w = nw;
   309→    h = nh;
   310→  }
   311→
   312→  const imgEsc = canvasDe(src, w, h);
   313→
   314→  // op3 — maybe_add_letterbox: h ≤ 30 o w/h > 8 → pads arriba/abajo negros.
   315→  //   padding_h = int(abs(max(int(w/8), 30) * 2 − h) / 2)   [main.py L161-164]
   316→  let padTop = 0;
   317→  let canvasPadded = imgEsc;
   318→  const widthHeightRatio = 8;
   319→  const minHeight = 30;
   320→  if (h <= minHeight || w / h > widthHeightRatio) {
   321→    const newH = Math.max(Math.floor(w / widthHeightRatio), minHeight) * 2;
   322→    padTop = Math.floor(Math.abs(newH - h) / 2);
   323→    const padded = canvasCon(w, h + padTop * 2); // SPRINT-A A3 — factory
   324→    const pctx = ctxDe(padded);
   325→    pctx.fillStyle = "#000000";
   326→    pctx.fillRect(0, 0, padded.width, padded.height);
   327→    pctx.drawImage(imgEsc, 0, padTop);
   328→    canvasPadded = padded;
   329→    h = padded.height;
   330→  }
   331→
   332→  // op4 — det-preprocess (limit <cap> max + round32 SIEMPRE, mínimo 32) —
   333→  //   cap=DET_CAP_PROD (1920) = producción de hoy byte a byte; 1280 = 1.ª
   334→  //   pasada del escalonado (banco B3: −56% det, 0 votos perdidos).
   335→  const ratioDet = Math.max(h, w) > cap ? cap / Math.max(h, w) : 1;
   336→  const dh = round32(Math.floor(h * ratioDet));
   337→  const dw = round32(Math.floor(w * ratioDet));
   338→  const detIn = canvasDe(canvasPadded, dw, dh);
   339→
   340→  return {
   341→    canvas: detIn,
   342→    ops: { w: dw, h: dh, ratioW, ratioH, padTop, ratioWDet: dw / w, ratioHDet: dh / h },
   343→  };
   344→}

   386→/** Convex hull (monotone chain) de un conjunto de puntos. */
   387→function convexHull(pts: { x: number; y: number }[]): { x: number; y: number }[] {
   388→  if (pts.length < 3) return pts.slice();
   389→  const p = pts.slice().sort((a, b) => (a.x === b.x ? a.y - b.y : a.x - b.x));
   390→  const cross = (o: { x: number; y: number }, a: { x: number; y: number }, b: { x: number; y: number }) =>
   391→    (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
   392→  const lower: { x: number; y: number }[] = [];
   393→  for (const q of p) {
   394→    while (lower.length >= 2 && cross(lower[lower.length - 2], lower[lower.length - 1], q) <= 0)
   395→      lower.pop();
   396→    lower.push(q);
   397→  }
   398→  const upper: { x: number; y: number }[] = [];
   399→  for (let i = p.length - 1; i >= 0; i--) {
   400→    const q = p[i];
   401→    while (upper.length >= 2 && cross(upper[upper.length - 2], upper[upper.length - 1], q) <= 0)
   402→      upper.pop();
   403→    upper.push(q);
   404→  }
   405→  lower.pop();
   406→  upper.pop();
   407→  return lower.concat(upper);
   408→}

   410→/** min-area rect por barrido de aristas del hull (proyectar sobre el eje de
   411→ *  cada arista y su perpendicular; quedarse con el área mínima). */
   412→function minAreaRect(hull: { x: number; y: number }[]): Rect | null {
   413→  if (hull.length < 2) return null;
   414→  let mejor: Rect | null = null;
   415→  let mejorArea = Infinity;
   416→  const n = hull.length;
   417→  for (let i = 0; i < n; i++) {
   418→    const p1 = hull[i];
   419→    const p2 = hull[(i + 1) % n];
   420→    const dx = p2.x - p1.x;
   421→    const dy = p2.y - p1.y;
   422→    const len = Math.hypot(dx, dy);
   423→    if (len < 1e-6) continue;
   424→    const ux = dx / len;
   425→    const uy = dy / len; // eje de la arista
   426→    let minU = Infinity;
   427→    let maxU = -Infinity;
   428→    let minV = Infinity;
   429→    let maxV = -Infinity;
   430→    for (const q of hull) {
   431→      const u = q.x * ux + q.y * uy;
   432→      const v = -q.x * uy + q.y * ux; // perpendicular n̂ = (−uy, ux)
   433→      if (u < minU) minU = u;
   434→      if (u > maxU) maxU = u;
   435→      if (v < minV) minV = v;
   436→      if (v > maxV) maxV = v;
   437→    }
   438→    const w = maxU - minU;
   439→    const h = maxV - minV;
   440→    const area = w * h;
   441→    if (area < mejorArea) {
   442→      mejorArea = area;
   443→      mejor = {
   444→        cx: ((maxU + minU) / 2) * ux + ((maxV + minV) / 2) * -uy,
   445→        cy: ((maxU + minU) / 2) * uy + ((maxV + minV) / 2) * ux,
   446→        w,
   447→        h,
   448→        theta: Math.atan2(uy, ux),
   449→      };
   450→    }
   451→  }
   452→  return mejor;
   453→}

   455→/** box_score_fast: media de `pred` dentro del quad (bbox + máscara por
   456→ *  punto-en-polígono, scanline por cruce de aristas). */
   457→function boxScoreFast(pred: Float32Array, w: number, h: number, box: { x: number; y: number }[]): number {
   458→  let xmin = w - 1;
   459→  let xmax = 0;
   460→  let ymin = h - 1;
   461→  let ymax = 0;
   462→  for (const q of box) {
   463→    xmin = Math.max(0, Math.min(xmin, Math.floor(q.x)));
   464→    xmax = Math.min(w - 1, Math.max(xmax, Math.ceil(q.x)));
   465→    ymin = Math.max(0, Math.min(ymin, Math.floor(q.y)));
   466→    ymax = Math.min(h - 1, Math.max(ymax, Math.ceil(q.y)));
   467→  }
   468→  if (xmax < xmin || ymax < ymin) return 0;
   469→  let suma = 0;
   470→  let cuenta = 0;
   471→  for (let y = ymin; y <= ymax; y++) {
   472→    // intersecciones de la scanline y con las aristas del polígono
   473→    const xs: number[] = [];
   474→    for (let i = 0; i < box.length; i++) {
   475→      const a = box[i];
   476→      const b = box[(i + 1) % box.length];
   477→      if (a.y === b.y) continue;
   478→      const yMin = Math.min(a.y, b.y);
   479→      const yMax = Math.max(a.y, b.y);
   480→      if (y + 0.5 >= yMin && y + 0.5 < yMax) {
   481→        xs.push(a.x + ((y + 0.5 - a.y) * (b.x - a.x)) / (b.y - a.y));
   482→      }
   483→    }
   484→    if (xs.length < 2) continue;
   485→    xs.sort((a, b) => a - b);
   486→    for (let k = 0; k + 1 < xs.length; k += 2) {
   487→      const x0 = Math.max(xmin, Math.ceil(xs[k] - 0.5));
   488→      const x1 = Math.min(xmax, Math.floor(xs[k + 1] - 0.5));
   489→      for (let x = x0; x <= x1; x++) {
   490→        if (x >= xs[k] && x <= xs[k + 1]) {
   491→          suma += pred[y * w + x];
   492→          cuenta++;
   493→        }
   494→      }
   495→    }
   496→  }
   497→  return cuenta === 0 ? 0 : suma / cuenta;
   498→}

   500→/** Esquinas de un rect expandido (centro + ángulo), ordenadas TL,TR,BR,BL
   501→ *  (convención get_mini_boxes: sort por x, pares por y). */
   502→function esquinasRect(r: Rect): { x: number; y: number }[] {
   503→  const cos = Math.cos(r.theta);
   504→  const sin = Math.sin(r.theta);
   505→  const nx = -sin;
   506→  const ny = cos;
   507→  const hw = r.w / 2;
   508→  const hh = r.h / 2;
   509→  const pts = [
   510→    { x: r.cx + hw * cos + hh * nx, y: r.cy + hw * sin + hh * ny },
   511→    { x: r.cx + hw * cos - hh * nx, y: r.cy + hw * sin - hh * ny },
   512→    { x: r.cx - hw * cos + hh * nx, y: r.cy - hw * sin + hh * ny },
   513→    { x: r.cx - hw * cos - hh * nx, y: r.cy - hw * sin - hh * ny },
   514→  ];
   515→  pts.sort((a, b) => a.x - b.x);
   516→  let tl: { x: number; y: number };
   517→  let bl: { x: number; y: number };
   518→  let tr: { x: number; y: number };
   519→  let br: { x: number; y: number };
   520→  if (pts[0].y <= pts[1].y) {
   521→    tl = pts[0];
   522→    bl = pts[1];
   523→  } else {
   524→    tl = pts[1];
   525→    bl = pts[0];
   526→  }
   527→  if (pts[2].y <= pts[3].y) {
   528→    tr = pts[2];
   529→    br = pts[3];
   530→  } else {
   531→    tr = pts[3];
   532→    br = pts[2];
   533→  }
   534→  return [tl, tr, br, bl];
   535→}

   537→/** sorted_boxes estándar PaddleOCR: por y superior; solapamiento vertical
   538→ *  (< 10 px) → por x (insert estabilizado del original). */
   539→function ordenarCajas(boxes: { x: number; y: number }[][]): { x: number; y: number }[][] {
   540→  const sorted = boxes.slice().sort((a, b) => a[0].y - b[0].y || a[0].x - b[0].x);
   541→  for (let i = 0; i < sorted.length - 1; i++) {
   542→    for (let j = i; j >= 0; j--) {
   543→      if (
   544→        Math.abs(sorted[j + 1][0].y - sorted[j][0].y) < 10 &&
   545→        sorted[j + 1][0].x < sorted[j][0].x
   546→      ) {
   547→        const tmp = sorted[j];
   548→        sorted[j] = sorted[j + 1];
   549→        sorted[j + 1] = tmp;
   550→      } else break;
   551→    }
   552→  }
   553→  return sorted;
   554→}

   558→function cropLinea(
   559→  src: HTMLImageElement,
   560→  box: { x: number; y: number }[],
   561→): { canvas: HTMLCanvasElement; ratio: number } {
   562→  const d = (a: { x: number; y: number }, b: { x: number; y: number }) => Math.hypot(b.x - a.x, b.y - a.y);
   563→  const [p0, p1, p2, p3] = box;
   564→  const cropW = Math.max(1, Math.round(Math.max(d(p0, p1), d(p2, p3))));
   565→  const cropH = Math.max(1, Math.round(Math.max(d(p0, p3), d(p1, p2))));
   566→  const theta = Math.atan2(p1.y - p0.y, p1.x - p0.x);
   567→  const cos = Math.cos(theta);
   568→  const sin = Math.sin(theta);
   569→  const canvas = canvasCon(cropW, cropH); // SPRINT-A A3 — factory
   570→  const ctx = ctxDe(canvas);
   571→  // get_rotate_crop_image equivalente: la matriz del contexto transforma
   572→  // coords de la FUENTE → coords del CANVAS (inversa de [R|p0]): rotación
   573→  // −θ y traslación −R·p0 — así la caja queda pegada al origen del canvas
   574→  // y la imagen se dibuja 1:1 (la ventana cropW×cropH recorta el rect).
   575→  // (v1 tenía el transform al revés: el dibujo caía fuera del canvas →
   576→  // crops transparentes y rec vacío — bug v1.)
   577→  const e = -cos * p0.x - sin * p0.y;
   578→  const f = sin * p0.x - cos * p0.y;
   579→  ctx.setTransform(cos, -sin, sin, cos, e, f);
   580→  ctx.imageSmoothingEnabled = true;
   581→  ctx.drawImage(src, 0, 0);
   582→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   583→  return { canvas, ratio: cropW / cropH };
   584→}

   586→function tensorDeCanvases(
   587→  ort: OrtModule,
   588→  canvases: HTMLCanvasElement[],
   589→  targetW: number,
   590→): OrtValue {
   591→  const b = canvases.length;
   592→  const data = new Float32Array(b * 3 * 48 * targetW);
   593→  for (let i = 0; i < b; i++) {
   594→    const c = canvases[i];
   595→    const ctx = ctxDe(c);
   596→    const { width: cw, height: ch } = c;
   597→    const img = ctx.getImageData(0, 0, cw, ch);
   598→    // resize PaddleOCR: h=48 fijo, ancho objetivo min(ceil(48·ratio), targetW)
   599→    // (pad de ceros a la derecha — el buffer ya nace en 0).
   600→    const rw = Math.min(Math.max(1, Math.ceil((48 * cw) / ch)), targetW);
   601→    for (let y = 0; y < 48; y++) {
   602→      const sy = Math.min(ch - 1, Math.floor((y * ch) / 48));
   603→      for (let x = 0; x < rw; x++) {
   604→        const sx = Math.min(cw - 1, Math.floor((x * cw) / rw));
   605→        const idx = (sy * cw + sx) * 4;
   606→        const o = i * 3 * 48 * targetW + y * targetW + x;
   607→        data[o] = (img.data[idx + 2] / 255 - 0.5) / 0.5; // B
   608→        data[o + 48 * targetW] = (img.data[idx + 1] / 255 - 0.5) / 0.5; // G
   609→        data[o + 2 * 48 * targetW] = (img.data[idx] / 255 - 0.5) / 0.5; // R
   610→      }
   611→    }
   612→  }
   613→  return new ort.Tensor("float32", data, [b, 3, 48, targetW]);
   614→}

   616→function softmax2(a: number, b: number): [number, number] {
   617→  const m = Math.max(a, b);
   618→  const ea = Math.exp(a - m);
   619→  const eb = Math.exp(b - m);
   620→  return [ea / (ea + eb), eb / (ea + eb)];
   621→}

   623→/** CTC decode: argmax por t; colapsar repetidos; ignorar blank (0);
   624→ *  1..6622 → dict[idx-1]; 6623 → " "; conf = media de las probs softmax de
   625→ *  los tokens seleccionados (CTCLabelDecode del paquete). */
   626→function ctcDecode(
   627→  tensor: OrtValue,
   628→  dict: string[],
   629→): { texto: string; conf: number } {
   630→  const dims = tensor.dims;
   631→  const T = dims[1];
   632→  const C = dims[2];
   633→  const data = tensor.data as Float32Array;
   634→  let texto = "";
   635→  let sumaConf = 0;
   636→  let cuenta = 0;
   637→  let ultimo = -1;
   638→  for (let t = 0; t < T; t++) {
   639→    const base = t * C;
   640→    let mejor = 0;
   641→    let mejorV = -Infinity;
   642→    for (let c = 0; c < C; c++) {
   643→      const v = data[base + c];
   644→      if (v > mejorV) {
   645→        mejorV = v;
   646→        mejor = c;
   647→      }
   648→    }
   649→    if (mejor !== 0 && mejor !== ultimo) {
   650→      // conf = prob del token: el ONNX del rec ya sale con softmax (igual que
   651→      // el paquete: preds.max(axis=2) sin re-normalizar — el doble softmax
   652→      // con shift aplastaba la conf a ~0 y descartaba TODAS las líneas).
   653→      const prob = mejorV;
   654→      if (mejor <= dict.length) texto += dict[mejor - 1];
   655→      else texto += " ";
   656→      sumaConf += prob;
   657→      cuenta++;
   658→    }
   659→    ultimo = mejor;
   660→  }
   661→  return { texto, conf: cuenta === 0 ? 0 : sumaConf / cuenta };
   662→}

   667→  boxesFuente: { x: number; y: number }[][];
   668→  msDet: number;
   669→  nBoxes: number;
   670→  /** D62 T2 — SOLO en HIT del memo: ms de det que esta llamada NO pagó

   672→  memoMsAhorrado?: number;

   675→// ─── KIT e14-probe-adaptativo-v1 (D62 T2): memo del det por identidad ────────
   676→// Cuando el probe NO aplica rotación (aplicado=0), el det a 1280 del probe
   677→// (rapidOcrCropsOrientacion) y el det a 1280 de la pasada-1 del escalonado
   678→// (rapidOcrReconocer con detCap=DET_CAP_RAPIDO) corren sobre el MISMO bitmap —
   679→// el segundo es gratis si se memoiza. Clave = el PROPIO string dataURL de
   680→// entrada (identidad exacta: rotar ⇒ re-encode ⇒ string distinto ⇒ miss
   681→// natural, cero ramas especiales) + cap (en producción 1920 la clave jamás
   682→// matchea la del probe 1280 ⇒ sin cambio de comportamiento). GUARD HONESTO:
   683→// el memo JAMÁS decide score/gate — devuelve el MISMO resultado que computaría
   684→// det (determinismo wasm: mismo bitmap + mismo cap ⇒ mismo boxesFuente/msDet;
   685→// se devuelve la MISMA referencia). Tope 2 entradas: insertar desplaza la más
   686→// vieja (probe y pasada-1 son consecutivos; nunca coexisten más). El memo es
   687→// solo de ejecución (no persiste) y no corre en SIM (el probe no corre allí —
   688→// guard del bridge; si otra vía alimenta el memo, devuelve lo mismo que
   689→// computaría — inofensivo).
   690→interface DetMemoEntrada {
   691→  cap: number;
   692→  boxesFuente: { x: number; y: number }[][];
   693→  msDet: number;
   694→  nBoxes: number;
   695→}
   696→
   697→const DET_MEMO_TOPE = 2;
   698→const detMemo = new Map<string, DetMemoEntrada>();
   699→
   700→function detMemoGuardar(clave: string, entrada: DetMemoEntrada): void {
   701→  if (detMemo.size >= DET_MEMO_TOPE) {
   702→    const masVieja = detMemo.keys().next().value; // 1.ª insertada = más vieja
   703→    if (masVieja !== undefined) detMemo.delete(masVieja);
   704→  }
   705→  detMemo.delete(clave); // si la clave ya estaba (otro cap), renueva posición
   706→  detMemo.set(clave, entrada);
   707→}

   709→async function inferDet(
   710→  rt: RapidRuntime,
   711→  src: HTMLImageElement,
   712→  cap: number = DET_CAP_PROD,
   713→  // D62 T2 — identidad del bitmap de origen (dataURL) para el memo; los 2
   714→  // callers la pasan (reconocer + cropsOrientacion). Ausente = sin memo.
   715→  claveOrigen?: string,
   716→): Promise<DetSalida> {
   717→  // D62 T2 — HIT: el probe ACABA de computar este mismo det (mismo dataURL +
   718→  // mismo cap, aplicado=0). Devuelve el MISMO resultado que computaría det —
   719→  // jamás decide score/gate. Los "det run" no se emiten (no corrió aquí).
   720→  if (claveOrigen != null) {
   721→    const hit = detMemo.get(claveOrigen);
   722→    if (hit !== undefined && hit.cap === cap) {
   723→      return {
   724→        boxesFuente: hit.boxesFuente,
   725→        msDet: hit.msDet,
   726→        nBoxes: hit.nBoxes,
   727→        memoMsAhorrado: hit.msDet,
   728→      };
   729→    }
   730→  }
   731→  const t0 = performance.now();
   732→  const { canvas, ops } = preDet(src, cap);
   733→  const ctx = ctxDe(canvas);
   734→  const { width: w, height: h } = canvas;
   735→  const img = ctx.getImageData(0, 0, w, h);
   736→  // PaddleOCR trabaja en BGR (cv2) — los ONNX del PP-OCRv4 esperan ese orden.
   737→  const data = new Float32Array(w * h * 3);
   738→  for (let i = 0, px = 0; i < img.data.length; i += 4, px++) {
   739→    data[px] = (img.data[i + 2] / 255 - 0.5) / 0.5; // B
   740→    data[w * h + px] = (img.data[i + 1] / 255 - 0.5) / 0.5; // G
   741→    data[2 * w * h + px] = (img.data[i] / 255 - 0.5) / 0.5; // R
   742→  }
   743→  const tensor = new rt.ort.Tensor("float32", data, [1, 3, h, w]);
   744→  logger.info(`[e14-rapidocr] det run [1,3,${h},${w}]...`);
   745→  const res = await rt.det.run({ [rt.det.inputNames[0]]: tensor });
   746→  logger.info(`[e14-rapidocr] det run OK (${Math.round(performance.now() - t0)}ms)`);
   747→  const out = res[rt.det.outputNames[0]];
   748→  const pred = out.data as Float32Array;
   749→  const pw = out.dims[3];
   750→  const ph = out.dims[2];
   751→  const msDet = Math.round(performance.now() - t0);
   752→
   753→  // 1) binarizar > 0.3
   754→  const bitmap = new Uint8Array(pw * ph);
   755→  for (let i = 0; i < bitmap.length; i++) bitmap[i] = pred[i] > DB_THRESH ? 1 : 0;
   756→
   757→  // 2) dilate 2×2 (max local con kernel [[1,1],[1,1]] — crece derecha/abajo)
   758→  const dil = new Uint8Array(pw * ph);
   759→  for (let y = 0; y < ph; y++) {
   760→    const fila = y * pw;
   761→    const filaPrev = Math.max(0, y - 1) * pw;
   762→    for (let x = 0; x < pw; x++) {
   763→      const xPrev = Math.max(0, x - 1);
   764→      const v = Math.max(
   765→        bitmap[fila + x],
   766→        bitmap[filaPrev + x],
   767→        bitmap[fila + xPrev],
   768→        bitmap[filaPrev + xPrev],
   769→      );
   770→      dil[fila + x] = v;
   771→    }
   772→  }
   773→
   774→  // 3) CCL 4-conectividad (BFS, máx 1000 componentes)
   775→  const labels = new Int32Array(pw * ph); // 0 = no visitado
   776→  let nComp = 0;
   777→  const rects: Rect[] = [];
   778→  const stack = new Int32Array(pw * ph);
   779→  const coords: number[] = [];
   780→  for (let y = 0; y < ph && nComp < DB_MAX_CANDIDATES; y++) {
   781→    for (let x = 0; x < pw && nComp < DB_MAX_CANDIDATES; x++) {
   782→      const idx = y * pw + x;
   783→      if (dil[idx] === 0 || labels[idx] !== 0) continue;
   784→      nComp++;
   785→      let sp = 0;
   786→      stack[sp++] = idx;
   787→      labels[idx] = nComp;
   788→      coords.length = 0;
   789→      while (sp > 0) {
   790→        const cur = stack[--sp];
   791→        const cy = (cur / pw) | 0;
   792→        const cx = cur - cy * pw;
   793→        coords.push(cx, cy);
   794→        // vecinos (4-con)
   795→        if (cx > 0 && dil[cur - 1] === 1 && labels[cur - 1] === 0) {
   796→          labels[cur - 1] = nComp;
   797→          stack[sp++] = cur - 1;
   798→        }
   799→        if (cx < pw - 1 && dil[cur + 1] === 1 && labels[cur + 1] === 0) {
   800→          labels[cur + 1] = nComp;
   801→          stack[sp++] = cur + 1;
   802→        }
   803→        if (cy > 0 && dil[cur - pw] === 1 && labels[cur - pw] === 0) {
   804→          labels[cur - pw] = nComp;
   805→          stack[sp++] = cur - pw;
   806→        }
   807→        if (cy < ph - 1 && dil[cur + pw] === 1 && labels[cur + pw] === 0) {
   808→          labels[cur + pw] = nComp;
   809→          stack[sp++] = cur + pw;
   810→        }
   811→      }
   812→      const nPx = coords.length / 2;
   813→      if (nPx < 10) continue; // ruido (findContours de <10 px tampoco aporta)
   814→      // submuestreo si > 20000 px
   815→      const stride = nPx > 20000 ? Math.ceil(nPx / 20000) : 1;
   816→      const pts: { x: number; y: number }[] = [];
   817→      for (let k = 0; k < nPx; k += stride) pts.push({ x: coords[k * 2], y: coords[k * 2 + 1] });
   818→      const hull = convexHull(pts);
   819→      const rect = minAreaRect(hull);
   820→      if (!rect) continue;
   821→      const sside = Math.min(rect.w, rect.h);
   822→      if (sside < MIN_SIDE_DESCARTAR) continue;
   823→      // esquinas del rect (pre-unclip) para el score
   824→      const esq = esquinasRect(rect);
   825→      const score = boxScoreFast(pred, pw, ph, esq);
   826→      if (score < DB_BOX_THRESH) continue;
   827→      // unclip (rect es exacto): d = área·ratio / perímetro; w += 2d, h += 2d
   828→      const perim = 2 * (rect.w + rect.h);
   829→      const dUnclip = (rect.w * rect.h * DB_UNCLIP_RATIO) / perim;
   830→      const wU = rect.w + 2 * dUnclip;
   831→      const hU = rect.h + 2 * dUnclip;
   832→      if (Math.min(wU, hU) < MIN_SIDE_TRAS_UNCLIP) continue;
   833→      rects.push({ cx: rect.cx, cy: rect.cy, w: wU, h: hU, theta: rect.theta });
   834→    }
   835→  }
   836→
   837→  // 4) esquinas TL,TR,BR,BL + mapeo a coords de la FUENTE
   838→  const boxesFuente = rects
   839→    .map((r) => esquinasRect(r).map((p) => mapearAFuente(p, ops, src.naturalWidth, src.naturalHeight)))
   840→    .filter((b) => b.every((p) => p.x >= 0 && p.y >= 0));
   841→  const salida: DetSalida = { boxesFuente: ordenarCajas(boxesFuente), msDet, nBoxes: rects.length };
   842→  // D62 T2 — MISS: alimenta el memo (tope 2; insertar desplaza la más vieja).
   843→  if (claveOrigen != null) detMemoGuardar(claveOrigen, { cap, ...salida });
   844→  return salida;
   845→}

   850→ * Secuencia completa det→cls→rec. KIT e14-motor-rapidocr-v1 (PROD-2 §4.3-C4):

   855→export async function rapidOcrReconocer(
   856→  rt: RapidRuntime,
   857→  dataUrl: string,
   858→  onProgress?: (p: number) => void,
   859→  // KIT e14-ocr-escalonado-v1 (§2.1-§2.2): cap det de ESTA llamada —
   860→  // ausente = DET_CAP_PROD (comportamiento de hoy byte a byte); la 1.ª
   861→  // pasada del escalonado pasa DET_CAP_RAPIDO. La sesión/cls/rec NO cambian.
   862→  opts?: { detCap?: number },
   863→): Promise<RapidOcrResultado> {
   864→  const tTotal = performance.now();
   865→  const src = await loadImageMemo(dataUrl); // SPRINT-A A2 — decode memoizado del repo
   866→  // D62 T2 — claveOrigen: el memo del det (probe 1280 ↔ pasada-1 1280 sobre el
   867→  // MISMO bitmap ⇒ HIT; aplicado=1 cambia el string ⇒ miss natural).

   869→

   871→    return { lineas: [], texto: "", confMedia: 0, msDet, msCls: 0, msRec: 0, nBoxes: 0 };
   872→  }
   873→
   874→  // 5) crops por línea (desde la FUENTE, resolución nativa)
   875→  const crops = boxesFuente.map((b) => cropLinea(src, b));
   876→
   877→  // 6) cls por lote de 6 (rotación 180 si score > 0.9)
   878→  const t1 = performance.now();
   879→  const rotar: boolean[] = new Array(crops.length).fill(false);
   880→  for (let i = 0; i < crops.length; i += 6) {
   881→    const lote = crops.slice(i, i + 6).map((c) => c.canvas);
   882→    const tensors = lote.map((cv) => {
   883→      // cls: h=48, w=min(192, ceil(48·ratio)) con pad a 192
   884→      const ratio = cv.width / cv.height;
   885→      const wObj = Math.min(192, Math.max(1, Math.ceil(48 * ratio)));
   886→      const tmp = canvasCon(wObj, 48); // SPRINT-A A3 — factory
   887→      ctxDe(tmp).drawImage(cv, 0, 0, wObj, 48);
   888→      return tmp;
   889→    });
   890→    const tensor = tensorDeCanvases(rt.ort, tensors, 192);
   891→    const res = await rt.cls.run({ [rt.cls.inputNames[0]]: tensor });
   892→    const out = res[rt.cls.outputNames[0]];
   893→    const data = out.data as Float32Array;
   894→    for (let k = 0; k < lote.length; k++) {
   895→      const [p0, p180] = out.dims[1] === 2 ? softmax2(data[k * 2], data[k * 2 + 1]) : [1, 0];
   896→      if (p180 > 0.9 && p180 > p0) rotar[i + k] = true;
   897→    }
   898→  }
   899→  const msCls = Math.round(performance.now() - t1);
   900→
   901→  // aplicar rotaciones (re-crop volteado 180°)
   902→  const cropsFinales = crops.map((c, i) => {
   903→    if (!rotar[i]) return c.canvas;
   904→    const cv = c.canvas;
   905→    const tmp = canvasCon(cv.width, cv.height); // SPRINT-A A3 — factory
   906→    const ctx = ctxDe(tmp);
   907→    ctx.translate(cv.width, cv.height);
   908→    ctx.rotate(Math.PI);
   909→    ctx.drawImage(cv, 0, 0);
   910→    return tmp;
   911→  });
   912→
   913→  // 7) rec por lote de 6 (líneas ordenadas por ratio w/h — acelera)
   914→  const t2 = performance.now();
   915→  const orden = cropsFinales
   916→    .map((c, i) => ({ i, ratio: c.width / c.height }))
   917→    .sort((a, b) => a.ratio - b.ratio);
   918→  const lineas: RapidOcrLinea[] = new Array(cropsFinales.length);
   919→  for (let k = 0; k < orden.length; k += 6) {
   920→    const lote = orden.slice(k, k + 6);
   921→    const maxRatio = Math.max(320 / 48, ...lote.map((o) => o.ratio));
   922→    // KIT e14-rec-wmax480-v1 (bench v8.5): cap del ancho del tensor — el lote
   923→    // entero corre al ancho de su línea más ancha; sin cap, una sola línea
   924→    // de 864 px hinchaba TODOS los tensores del lote (0.56 ms/unidad WASM).
   925→    const W = Math.min(REC_WMAX, Math.floor(48 * maxRatio));
   926→    const canvases = lote.map((o) => cropsFinales[o.i]);
   927→    const tensor = tensorDeCanvases(rt.ort, canvases, W);
   928→    const res = await rt.rec.run({ [rt.rec.inputNames[0]]: tensor });
   929→    const out = res[rt.rec.outputNames[0]];
   930→    for (let m = 0; m < lote.length; m++) {
   931→      const dims = out.dims;
   932→      const sub: OrtValue = {
   933→        dims: [1, dims[1], dims[2]],
   934→        data: (out.data as Float32Array).subarray(
   935→          m * dims[1] * dims[2],
   936→          (m + 1) * dims[1] * dims[2],
   937→        ) as Float32Array,
   938→        type: out.type,
   939→      };
   940→      const { texto, conf } = ctcDecode(sub, rt.dict);
   941→      if (conf >= TEXT_SCORE && texto.trim().length > 0) {
   942→        lineas[lote[m].i] = { texto: texto.trim(), conf, box: boxesFuente[lote[m].i] };
   943→      }
   944→    }
   945→    // PROD-2 (§4.3-C4): progreso real por lotes de rec — i/total (0→1).
   946→    onProgress?.(Math.min(1, (k + lote.length) / orden.length));
   947→  }
   948→  const msRec = Math.round(performance.now() - t2);
   949→
   950→  const aceptadas = lineas.filter((l): l is RapidOcrLinea => !!l);
   951→  const texto = aceptadas.map((l) => l.texto).join("\n");
   952→  const confMedia =
   953→    aceptadas.length === 0 ? 0 : aceptadas.reduce((s, l) => s + l.conf, 0) / aceptadas.length;
   954→  logger.info(
   955→    `[e14-rapidocr] celda: det=${msDet}ms cls=${msCls}ms rec=${msRec}ms · ${boxesFuente.length} cajas · ${aceptadas.length} líneas · conf ${(confMedia * 100).toFixed(1)}% · ${Math.round(performance.now() - tTotal)}ms detMemo=${memoMsAhorrado != null ? `hit msAhorrado=${memoMsAhorrado}` : "miss"}`,
   956→  );
   957→  return {
   958→    lineas: aceptadas,
   959→    texto,
   960→    confMedia,
   961→    msDet,
   962→    msCls,
   963→    msRec,
   964→    nBoxes: boxesFuente.length,
   965→  };
   966→}

   985→export async function rapidOcrCropsOrientacion(
   986→  rt: RapidRuntime,
   987→  dataUrl: string,
   988→  k: number,
   989→): Promise<{ crops: HTMLCanvasElement[]; w: number; h: number; msDet: number }> {
   990→  const src = await loadImageMemo(dataUrl); // SPRINT-A A2 — decode memoizado del repo
   991→  // D56 §3 T2: det del probe a DET_CAP_RAPIDO (banco B3 GO: −56%, 0 cajas
   992→  // perdidas) — el probe no necesita det de producción, sí caber en presupuesto.
   993→  // D62 T2 — claveOrigen: este det ALIMENTA el memo (mismo bitmap + cap 1280
   994→  // ⇒ la pasada-1 del escalonado con aplicado=0 hace HIT y no repite el det).
   995→  const { boxesFuente, msDet } = await inferDet(rt, src, DET_CAP_RAPIDO, dataUrl);
   996→  const n = boxesFuente.length;
   997→  const crops: HTMLCanvasElement[] = [];
   998→  if (n > 0) {
   999→    const kk = Math.min(k, n);
  1000→    const largoDe = (b: { x: number; y: number }[]) =>
  1001→      Math.max(
  1002→        Math.hypot(b[1].x - b[0].x, b[1].y - b[0].y),
  1003→        Math.hypot(b[2].x - b[1].x, b[2].y - b[1].y),
  1004→      );
  1005→    const nAnchas = Math.min(3, kk);
  1006→    const anchas = [...boxesFuente]
  1007→      .sort((a, b) => largoDe(b) - largoDe(a))
  1008→      .slice(0, nAnchas);
  1009→    const enAnchas = new Set(anchas);
  1010→    const resto = [...boxesFuente]
  1011→      .sort(
  1012→        (a, b) => a.reduce((s, p) => s + p.y, 0) - b.reduce((s, p) => s + p.y, 0),
  1013→      )
  1014→      .filter((b) => !enAnchas.has(b));
  1015→    const faltan = kk - anchas.length;
  1016→    const paso = resto.length / Math.max(1, faltan); // ≥ 1 → índices sin repetir
  1017→    const elegidas = [...anchas];
  1018→    for (let i = 0; i < faltan; i++) {
  1019→      elegidas.push(resto[Math.floor(i * paso)]);
  1020→    }
  1021→    for (const b of elegidas) crops.push(cropLinea(src, b).canvas);
  1022→  }
  1023→  return { crops, w: src.naturalWidth, h: src.naturalHeight, msDet };
  1024→}
