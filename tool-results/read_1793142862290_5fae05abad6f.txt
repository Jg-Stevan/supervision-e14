   127→/** Veredicto del pre-scan para UNA fila (§D.2 — ESPEJO de cadenaDedup:

   129→export type VeredictoPrescan =
   130→  | { accion: "duplicada"; originalId: string }
   131→  | { accion: "ya-enviada"; enviadaId: string }
   132→  | { accion: "registrar" } // payload NUEVO → registrarQrPayload + sigue PENDIENTE
   133→  | { accion: "nada" }; // sin payload / conservador → sigue PENDIENTE

   135→export interface DepsClasificadorPrescan {
   136→  originalDePayload: (p: string) => string | null;
   137→  /** existe en filas (viva — QUITADA/vaciada ⇒ false). */
   138→  filaViva: (id: string) => boolean;
   139→  enviadaPorPayload: (p: string) => Promise<{ id: string } | null>;
   140→}
   141→
   142→/**
   143→ * Orden de decisión (kit §D.2 — literal):
   144→ *  1. payload null → `nada` (los 4 arrugados pasan al flujo normal — regla
   145→ *     de oro 4).
   146→ *  2. original = originalDePayload(payload); si existe, ≠ filaId y la fila
   147→ *     original está VIVA → `duplicada` (el reuso clona el desenlace — jamás
   148→ *     doble envío).
   149→ *     Guard transversal: match a fila NO viva → `nada` (conservador: OCR
   150→ *     decidirá — jamás marcar DUPLICADA apuntando a un zombi).
   151→ *  3. Si no: enviadaPorPayload(payload) → `ya-enviada` (via "payload") —
   152→ *     mismo self-guard de la capa 0 (runner.ts:567: hit.id !== fila.id).
   153→ *  4. Si no: `registrar`.
   154→ */
   155→export async function clasificarPayloadPrescan(
   156→  payload: string | null,
   157→  filaId: string,
   158→  deps: DepsClasificadorPrescan,
   159→): Promise<VeredictoPrescan> {
   160→  // 1 · sin payload — los arrugados/giros pasan al OCR (jamás descartar)
   161→  if (payload == null || payload.length === 0) return { accion: "nada" };
   162→  // 2 · dup EN LOTE (mapaQrPayload — capas 0 del runner ven lo mismo)
   163→  const original = deps.originalDePayload(payload);
   164→  if (original != null && original !== filaId) {
   165→    if (deps.filaViva(original)) return { accion: "duplicada", originalId: original };
   166→    return { accion: "nada" }; // guard transversal — conservador: OCR decidirá
   167→  }
   168→  // 3 · histórico de ENVIADAS (ledger persistente — hit ⇒ YA-ENVIADA)
   169→  const hit = await deps.enviadaPorPayload(payload);
   170→  if (hit && hit.id !== filaId) return { accion: "ya-enviada", enviadaId: hit.id };
   171→  // 4 · payload NUEVO — se registra (la capa 0 del runner lo verá igual)
   172→  return { accion: "registrar" };
   173→}

   188→export interface DepsEscrituraPrescan {

   196→/**
   197→ * Escribe el veredicto con la MISMA maquinaria de marcas de siempre (§C):
   198→ *  · duplicada → write IDB (patrón cadenaDedup store.ts:586-594) + patch
   199→ *    SOLO-VISUAL. §F.2: el write OMITE el campo `dedup` (opcional) —
   200→ *    estado:"DUPLICADA" + dupOf bastan para reuso/UI/informe; el pre-scan
   201→ *    NO miente telemetría ajena (sin token "qr-prescan" en la unión §G).
   202→ *  · ya-enviada → marcarYaEnviada COMPLETA (guard anti-zombi + write IDB +
   203→ *    patchFila + alimenta el ledger) con via "payload" (§C).
   204→ *  · registrar → registrarQrPayload(payload, id) — la capa 0 del runner lo
   205→ *    verá igual en su re-chequeo.
   206→ *  · nada → no-op (queda PENDIENTE — Fase 3 lo procesa).
   207→ * El CALLER ya garantizó fila VIVA y PENDIENTE en el instante previo al
   208→ * write (anti-zombi §G); marcarYaEnviada re-verifica su guard por dentro.
   209→ */
   210→export async function aplicarVeredictoPrescan(
   211→  filaId: string,
   212→  veredicto: VeredictoPrescan,
   213→  payload: string | null,
   214→  deps: DepsEscrituraPrescan,
   215→): Promise<void> {
   216→  switch (veredicto.accion) {
   217→    case "duplicada": {
   218→      try {
   219→        const reg = await deps.leerRegistro(filaId);
   220→        await deps.guardarRegistro({
   221→          ...reg,
   222→          estado: "DUPLICADA",
   223→          dupOf: veredicto.originalId,
   224→          // SIN campo `dedup` (§F.2) — es opcional; el comportamiento queda
   225→          // completo (reuso <1 s, UI, informe) y el contrato D57 intacto.
   226→        });
   227→      } catch {
   228→        /* IDB caída / fila fuera de la cola → patch solo en memoria */
   229→      }
   230→      deps.actualizarFila(filaId, { estado: "DUPLICADA", dupOf: veredicto.originalId });
   231→      return;
   232→    }
   233→    case "ya-enviada":
   234→      // §C — marcarYaEnviada TAL CUAL (via "payload"): su guard interno
   235→      // exige PENDIENTE (anti-zombi doble) y alimenta el ledger (RL-e65b-2).
   236→      await deps.marcarYaEnviada(
   237→        filaId,
   238→        veredicto.enviadaId,
   239→        "payload",
   240→        undefined,
   241→        undefined,
   242→        payload ?? undefined,
   243→      );
   244→      return;
   245→    case "registrar":
   246→      if (payload) deps.registrarQrPayload(payload, filaId);
   247→      return;
   248→    case "nada":
   249→      return; // queda PENDIENTE — Fase 3 lo procesa
   250→  }
   251→}

   253→// ─── Lectura de QR sobre el warp — EXACTAMENTE la cadena del harness ────────

   255→/** Declaración local mínima — el repo NO declara BarcodeDetector (la red
   256→ *  line de identidad-qr.ts:54 lo PROHIBE para la capa 0/identidad y ese
   257→ *  archivo queda INTACTO). §F.1: aquí lo ordena el dueño para este módulo
   258→ *  NUEVO y es seguro por igualdad EXACTA de payload (Reed-Solomon — no hay
   259→ *  «casi el mismo payload»; el error siempre cae del lado conservador). */
   260→interface BarcodeDetectorLike {
   261→  detect(source: ImageBitmapSource): Promise<{ rawValue: string }[]>;
   262→}
   263→declare const BarcodeDetector:
   264→  | (new (opts?: { formats?: string[] }) => BarcodeDetectorLike)
   265→  | undefined;
   266→
   267→/** Cap del harness para rasterizar el warp en el fallback jsQR. */
   268→const QR_WARP_LADO_TOPE = 1600;
   269→
   270→/**
   271→ * (a) BarcodeDetector nativo → (b) jsQR sobre los píxeles del warp
   272→ * (intentoSobrePixeles — tryBoth ya viene adentro) → (c) null. Sin throw
   273→ * jamás (doctrina del conservador).
   274→ */
   275→export async function leerQrDeWarped(warped: ImageBitmap): Promise<string | null> {
   276→  try {
   277→    if (typeof BarcodeDetector !== "undefined") {
   278→      try {
   279→        const bd = new BarcodeDetector({ formats: ["qr_code"] });
   280→        const hits = await bd.detect(warped);
   281→        if (hits.length > 0 && hits[0]!.rawValue) return hits[0]!.rawValue;
   282→      } catch {
   283→        /* cae a jsQR */
   284→      }
   285→    }
   286→    if (typeof OffscreenCanvas === "undefined") return null;
   287→    const lado = Math.max(warped.width, warped.height);
   288→    const escala = lado > QR_WARP_LADO_TOPE ? QR_WARP_LADO_TOPE / lado : 1;
   289→    const w = Math.max(1, Math.round(warped.width * escala));
   290→    const h = Math.max(1, Math.round(warped.height * escala));
   291→    const canvas = new OffscreenCanvas(w, h);
   292→    const ctx = canvas.getContext("2d");
   293→    if (!ctx) return null;
   294→    ctx.drawImage(warped, 0, 0, w, h);
   295→    const img = ctx.getImageData(0, 0, w, h);
   296→    const pix: PixelesQr = { data: img.data, width: w, height: h };
   297→    return intentoSobrePixeles(pix, BANDA_A);
   298→  } catch {
   299→    return null; // (c) sin throw jamás
   300→  }
   301→}

   413→  );
   414→  // R2 — cola ya corriendo (iniciar() es singleton no-op, igual hoy)
   415→  if (store.corriendo) {

   417→    return;

   420→  // KIT e14-quad-streaming-v1 (P5): los kill-switches y el TURNO se

   422→  // el snapshot del objetivo y la marca del turno ven la MISMA tienda:

   432→  const prioridadOn = leerFlags().masivaPrioridad === true;
   433→  if (streamingOn) abrirTurnoClasificacion(Date.now());
   434→  // KIT flujo-dueno-v1 (K2): el warp de INGESTA posee sus filas — el objetivo
   435→  // de este pase EXCLUYE las que están en SU cola (encoladas o en vuelo: su

   437→  // ASEGURADOR — procesa filas SIN stamps y SIN dueño (agregadas mientras
   438→  // corría la corrida o filas que la ingesta no pudo decidir). Fuera del
   439→  // guard (flag OFF) el objetivo es el de hoy byte-idéntico.

   441→  const objetivo = objetivoPrescan(useMasivaStore.getState().filas).filter((f) =>
   442→    ingestaOn ? !filaBajoWarpIngesta(f.id) : true,
   443→  );
   444→  if (objetivo.length === 0) {
   445→    if (streamingOn) cerrarTurnoClasificacion(); // sin objetivo — turno inerte
   446→    // DRY RUN: nada que sondear ⇒ cola intacta (el pool JAMÁS arranca).
   447→    if (!soloWarpQr) store.iniciar(); // idempotente, como hoy
   448→    return;
   449→  }
   450→  if (streamingOn) {
   451→    // RL-e65b-6 — el escape del dueño es sagrado: las PROCESAR IGUAL van
   452→    // DIRECTO al pool sin pre-scan (stamp upfront — sin carrera posible:
   453→    // nadie las va a clasificar jamás).
   454→    for (const f of useMasivaStore.getState().filas) {
   455→      if (f.estado === "PENDIENTE" && f.forzarDedupOff) {
   456→        useMasivaStore.getState().actualizarFila(f.id, { prescanOk: true });
   457→      }
   458→    }
   459→  }
   460→  corriendoModulo = true;
   461→  emitir({ corriendo: true, total: objetivo.length, listo: 0, leidos: 0, fase: "boot" });
   462→  let leidos = 0;

   464→  try {
   465→    if (!(await cliente.listo())) {
   466→      // Abort honesto (§G): worker no bootea → flujo de HOY byte-idéntico.
   467→      logger.info("[e14] prescan-qr: lector no disponible — flujo de hoy (abort honesto)");
   468→    } else {
   469→      emitir({ fase: "sondeo" });
   470→      // R5 — FASE 1+2: por fila, EN ORDEN FIFO (for...of, serial, RAM móvil)
   471→      let procesadas = 0;
   472→      for (const fila of objetivo) {
   473→        emitir({ listo: procesadas, leidos }); // «SONDEANDO QR… k/N»
   474→        try {
   475→          // P1 (brief Agente A) — flag masivaQrHandoff: si el sondeo de
   476→          // ingesta (e73) YA leyó el QR de esta fila, TODO el bloque de
   477→          // lectura de abajo (decode FULL + OpenCV + warp + BarcodeDetector/
   478→          // jsQR) existía SOLO para obtener este payload → clasificar directo
   479→          // y saltar (el quad no viaja — eso es P4, otro parche). Anti-zombi

   481→          // del write); flag OFF / sin payload estampado → camino de hoy byte
   482→          // a byte. El estampado en mapaQrPayload lo hace aplicarVeredicto
   483→          // (veredicto "registrar") — el mapa queda como siempre.
   484→          const heredado = leerFlags().masivaQrHandoff
   485→            ? qrPayloadDeFilaStore(fila.id)
   486→            : null;
   487→          if (heredado != null) {
   488→            const vivaH = useMasivaStore
   489→              .getState()
   490→              .filas.find((f) => f.id === fila.id);
   491→            if (vivaH && vivaH.estado === "PENDIENTE") {
   492→              const veredictoH = await clasificarPayloadPrescan(
   493→                heredado,
   494→                fila.id,
   495→                DEPS_CLASIFICADOR,
   496→              );
   497→              await aplicarVeredictoPrescan(fila.id, veredictoH, heredado, DEPS_ESCRITURA);

   501→              );

   503→              // TAMBIÉN es «fila decidida» — stamp prescanOk AQUÍ, no en el

   511→                .getState()

   522→                  // P5 — arranque temprano (dry run JAMÁS arranca el pool).

   573→                    }

   578→          } finally {

   636→            }

   638→        } catch {

   640→        } finally {
   641→          procesadas++;
   642→        }
   643→      }
   644→      emitir({ listo: procesadas, leidos });

   672→        }
   673→      }
   674→      cerrarTurnoClasificacion();
   675→    }

   683→  if (!soloWarpQr) {
   684→    useMasivaStore.getState().iniciar();

   689→  }
