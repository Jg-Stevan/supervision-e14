   105→function relojDeMinutos(minutos: number): string {
   106→  const m = ((minutos % 1440) + 1440) % 1440;
   107→  const h24 = Math.floor(m / 60);

   109→  const sufijo = h24 >= 12 ? "PM" : "AM";
   110→  const h12 = h24 % 12 === 0 ? 12 : h24 % 12;

   379→/** Estado de mesa derivado de sus partes (regla única, D13). */
   380→function estadoMesaDePartes(mesa: Mesa): Mesa["estadoMesa"] {
   381→  if (
   382→    mesa.transmision.p1.alerta ||
   383→    mesa.transmision.p2.alerta
   384→  ) {
   385→    return "INCOMPLETO";
   386→  }
   387→  const partesOk =
   388→    (mesa.delegados.p1 === "ok" ? 1 : 0) +
   389→    (mesa.delegados.p2 === "ok" ? 1 : 0) +
   390→    (mesa.transmision.p1.estado === "ok" ? 1 : 0) +
   391→    (mesa.transmision.p2.estado === "ok" ? 1 : 0);
   392→  if (partesOk === 4) return "COMPLETO";
   393→  if (partesOk === 0) return "NO_INICIADO";
   394→  return "EN_PROCESO";
   395→}

   403→/** pcts derivables de las mesas (≈ % de partes ok, kit T2). */
   404→function pctsDeMesas(mesas: Mesa[]): { delegado: number; transmision: number } {
   405→  const total = mesas.length * 2;
   406→  if (total === 0) return { delegado: 0, transmision: 0 };
   407→  const delegOk = mesas.reduce(
   408→    (s, m) =>
   409→      s + (m.delegados.p1 === "ok" ? 1 : 0) + (m.delegados.p2 === "ok" ? 1 : 0),
   410→    0
   411→  );
   412→  const transOk = mesas.reduce(
   413→    (s, m) =>
   414→      s +
   415→      (m.transmision.p1.estado === "ok" ? 1 : 0) +
   416→      (m.transmision.p2.estado === "ok" ? 1 : 0),
   417→    0
   418→  );
   419→  return {
   420→    delegado: Math.round((delegOk / total) * 100),
   421→    transmision: Math.round((transOk / total) * 100),
   422→  };
   423→}

   544→/**
   545→ * KIT carga-masiva-supervision-v1 (D27) — superpone el registro RUNTIME del
   546→ * digitalizador (fotos + encabezado impreso) sobre el acta base del mock.
   547→ * Las fotos del runtime MANDAN en las páginas que traen; el resto cae al
   548→ * acta base o al sintético D18. Si la mesa no tenía acta (NO_INICIADO), se
   549→ * construye una mínima honesta desde el registro runtime.
   550→ */
   551→function fusionarActaRuntime(
   552→  base: ActaMesa | null,
   553→  runtime: import("./actas-reales/actas-reales").ActaRealLista,
   554→  puesto: Puesto,
   555→  mesa: Mesa
   556→): ActaMesa {
   557→  const horaRuntime = (() => {
   558→    const d = new Date();
   559→    return `${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")} LOCAL`;
   560→  })();
   561→  const acta: ActaMesa = base
   562→    ? {
   563→        ...base,
   564→        formularios: { ...base.formularios },
   565→        historial: [...base.historial],
   566→        fotos: { ...(base.fotos ?? {}) },
   567→      }
   568→    : {
   569→        id: `#MSV-${runtime.idEnlace.replace(/_/g, "")}-${runtime.mesa}`,
   570→        mesaId: mesa.id,
   571→        puestoId: puesto.id,
   572→        horaCarga: horaRuntime,
   573→        formularios: { TRANSMISION: [], DELEGADOS: [] },
   574→        historial: [
   575→          {
   576→            hora: horaRuntime,
   577→            titulo: "CARGA MASIVA",
   578→            detalle: "Acta recibida del digitalizador (carga masiva)",
   579→            nivel: "NEUTRAL",
   580→          },
   581→        ],
   582→        version: 1,
   583→      };
   584→
   585→  // Formularios: tamaño y estado por página — runtime con foto → VALIDA;
   586→  // sin foto → lo que diga el base, o PENDIENTE.
   587→  const formularios: Record<FormularioTipo, PaginaActa[]> = {
   588→    TRANSMISION: [...acta.formularios.TRANSMISION],
   589→    DELEGADOS: [...acta.formularios.DELEGADOS],
   590→  };
   591→  for (const formulario of ["TRANSMISION", "DELEGADOS"] as FormularioTipo[]) {
   592→    const slotRuntime = runtime.fotos[formulario];
   593→    if (!slotRuntime) continue;
   594→    const total = Math.max(
   595→      slotRuntime.length,
   596→      formularios[formulario].length,
   597→      1
   598→    );
   599→    const paginas: PaginaActa[] = [];
   600→    for (let i = 0; i < total; i++) {
   601→      paginas.push({
   602→        numero: i + 1,
   603→        estado: slotRuntime[i]
   604→          ? "VALIDA"
   605→          : (formularios[formulario][i]?.estado ?? "PENDIENTE"),
   606→      });
   607→    }
   608→    formularios[formulario] = paginas;
   609→  }
   610→  acta.formularios = formularios;
   611→
   612→  // Fotos: el runtime manda donde trae foto; el base cubre el resto.
   613→  const fotos = { ...(acta.fotos ?? {}) } as NonNullable<ActaMesa["fotos"]>;
   614→  for (const formulario of ["TRANSMISION", "DELEGADOS"] as FormularioTipo[]) {
   615→    const slotRuntime = runtime.fotos[formulario];
   616→    if (!slotRuntime) continue;
   617→    const baseSlot = fotos[formulario] ?? [];
   618→    const total = Math.max(slotRuntime.length, baseSlot.length);
   619→    const fusion: (import("../domain/acta").FotoActa | null)[] = [];
   620→    for (let i = 0; i < total; i++) {
   621→      fusion.push(slotRuntime[i] ?? baseSlot[i] ?? null);
   622→    }
   623→    fotos[formulario] = fusion;
   624→  }
   625→  acta.fotos = fotos;
   626→
   627→  // Encabezado impreso verbatim del runtime (se muestra, no se interpreta).
   628→  acta.impresion = runtime.impresion;
   629→  return acta;
   630→}

   632→function coincide(puesto: Puesto, f: FiltrosPuestos | undefined): boolean {
   633→  if (!f) return true;
   634→  if (f.pais && puesto.codigoPais !== f.pais) return false;
   635→  // KIT monitor-jornada-v1: filtro por día de la jornada electoral.
   636→  if (f.fecha && puesto.diaJornada !== f.fecha) return false;
   637→  if (f.zona && puesto.zonaId !== f.zona) return false;
   638→  if (f.puesto && puesto.puesto !== f.puesto) return false;
   639→  if (f.estado && puesto.estadoGlobal !== f.estado) return false;
   640→  const q = f.busqueda?.trim().toLowerCase();
   641→  if (q) {
   642→    const texto = [
   643→      puesto.id,
   644→      puesto.codigoPais,
   645→      puesto.pais,
   646→      puesto.zona,
   647→      puesto.puesto,
   648→    ]
   649→      .join(" ")
   650→      .toLowerCase();
   651→    if (!texto.includes(q)) return false;
   652→  }
   653→  return true;
   654→}
