   271→ * presupuesto de reló 4 000 ms (ZXING_BARRIDO_PRESUPUESTO_MS — chequeo

   274→ */
   275→export async function barridoQrZxingDePixeles(
   276→  pix: PixelesQr,
   277→): Promise<string | null> {
   278→  const mod = await moduloZxingDe();
   279→  if (!mod || typeof ImageData === "undefined") return null;
   280→  const limite = performance.now() + ZXING_BARRIDO_PRESUPUESTO_MS;
   281→  for (const q of CUADRANTES) {
   282→    const lim = limitesDeCuadrante(pix, q);
   283→    for (const fraccion of BARRIDO_VENTANAS) {
   284→      const lado = ladoDeVentana(pix, fraccion);
   285→      const paso = Math.max(1, Math.round(lado * 0.5));
   286→      for (let y = lim.y0; y + lado <= lim.y1; y += paso) {
   287→        for (let x = lim.x0; x + lado <= lim.x1; x += paso) {
   288→          if (performance.now() > limite) return null; // presupuesto vencido
   289→          const win = ventanaDePixeles(pix, x, y, lado);
   290→          if (!win) continue;
   291→          for (const binarizador of ZXING_BINARIZADORES) {
   292→            const payload = await zxingDeVentana(win, binarizador);
   293→            if (payload != null) return payload;
   294→          }
   295→        }
   296→      }
   297→      // e64-qr T-1.0 — celda final del cuadrante (hueco del filo, doc
   298→      // hermano H-5): anclada al filo inferior-derecho, MISMA grilla, los
   299→      // DOS binarizadores y el chequeo de presupuesto intacto.
   300→      if (performance.now() > limite) return null; // presupuesto vencido
   301→      const winFilo = ventanaDePixeles(pix, lim.x1 - lado, lim.y1 - lado, lado);
   302→      if (winFilo) {
   303→        for (const binarizador of ZXING_BINARIZADORES) {
   304→          const payloadFilo = await zxingDeVentana(winFilo, binarizador);
   305→          if (payloadFilo != null) return payloadFilo;
   306→        }
   307→      }
   308→    }
   309→  }
   310→  return null;
   311→}

   392→    const sh = Math.max(1, Math.round(bitmap.height * QR_ZOOM_FRAC_ALTO));
   393→    if (sw < 1 || sh < 1) return null;
   394→    const escala = Math.min(1, ladoDestino / Math.max(sw, sh));
   395→    const dw = Math.max(1, Math.round(sw * escala));

   398→      null;

   408→    ctx.imageSmoothingQuality = "medium";

   410→    const imgData = ctx.getImageData(0, 0, dw, dh);
   411→    return { data: imgData.data, width: dw, height: dh };

   413→    return null; // un fallo de render es sin-payload (fallback limpio)

   419→    }
   420→  }

   490→ * sigue directo al barrido zxing (con SU presupuesto propio de 4 s) y a

   495→export async function decodificarQrDePixeles(
   496→  pix: PixelesQr,
   497→  opciones: { presupuestoMs?: number | null } = {},
   498→): Promise<string | null> {
   499→  const t0 = Date.now();
   500→  const presupuestoMs =
   501→    typeof opciones.presupuestoMs === "number" && opciones.presupuestoMs > 0
   502→      ? opciones.presupuestoMs
   503→      : null;
   504→  const vencido = () => presupuestoMs !== null && Date.now() - t0 > presupuestoMs;
   505→  const base = ladderBaseDePixeles(pix);
   506→  if (base != null) {
   507→    trazaQr("base", base, t0);
   508→    return base;
   509→  }
   510→  // e71 (Opt-2) — tier RECORTE (ANTES de cualquier barrido completo): UNA
   511→  // lectura jsQR + zxing sobre la esquina estándar del E-14 (<500 ms). Si el
   512→  // QR vive donde el formulario lo pone (el caso de campo), los barridos
   513→  // completos (~15 s) JAMÁS corren. Fallback sagrado: el barrido de SIEMPRE
   514→  // sigue detrás — los QR fuera de la esquina estándar (actas rotadas u
   515→  // encuadres raros) conservan SU rescate, cero pérdida de cobertura.
   516→  const recorte = await decodificarQrDeRecorte(pix);
   517→  if (recorte != null) {
   518→    trazaQr("recorte-600", recorte, t0);
   519→    return recorte;
   520→  }
   521→  // KIT e14-gama-adaptativa-v1 — guard de presupuesto por tier pesado (las
   522→  // líneas `const barrido/nitido/binarizada = barridoQrDePixeles(...)` se
   523→  // conservan byte-idénticas — ancla source-scan qa-qr-identidad §T-1).
   524→  // R1.1 — KIT qr-presupuesto-real-v1: el guard de reló se INYECTA al barrido
   525→  // (activarGuardBarrido — chequeo POR CUADRANTE y POR FRACCIÓN dentro de
   526→  // barridoQrDePixeles): el presupuesto deja de saltarse a sí mismo al
   527→  // arrancar un barrido con el reló a punto de vencer. Con presupuesto null
   528→  // (gama BASE/ALTO) vencido() es siempre false ⇒ byte-idéntico a hoy.
   529→  activarGuardBarrido(vencido);
   530→  try {
   531→    if (!vencido()) {
   532→      const barrido = barridoQrDePixeles(pix);
   533→      if (barrido != null) {
   534→        trazaQr("barrido", barrido, t0);
   535→        return barrido;
   536→      }
   537→    } else {
   538→      logger.info(
   539→        `[e14] qr-capa0 presupuesto=${presupuestoMs}ms vencido tras recorte — salta barridos jsQR (gama adaptativa)`,
   540→      );
   541→    }
   542→    // e64-qr T-1.1 — tier NITIDEZ: barrido sobre la imagen enfocada (doc
   543→    // hermano R-1 — el remuestreo a 2048 suaviza los módulos; convolución
   544→    // ~25 ms a 2048² + barrido acotado por nº de ventanas).
   545→    if (!vencido()) {
   546→      const nitido = barridoQrDePixeles(nitidezQrDePixeles(pix));
   547→      if (nitido != null) {
   548→        trazaQr("barrido-nitido", nitido, t0);
   549→        return nitido;
   550→      }
   551→    }
   552→    // e64-qr T-1.2 — tier BINARIZADA: barrido sobre el blanco-negro con
   553→    // umbral LOCAL (doc hermano R-2 — sombra/iluminación dura donde el
   554→    // binarizado interno de jsQR falla; imagen integral O(1) por píxel).
   555→    if (!vencido()) {
   556→      const binarizada = barridoQrDePixeles(binarizadaQrDePixeles(pix));
   557→      if (binarizada != null) {
   558→        trazaQr("barrido-binarizado", binarizada, t0);
   559→        return binarizada;
   560→      }
   561→    }
   562→    const zxing = await barridoQrZxingDePixeles(pix);
   563→    if (zxing != null) {
   564→      trazaQr("zxing", zxing, t0);
   565→      return zxing;
   566→    }
   567→    // R1.1 — guard POR ROTACIÓN (antes corrían las 3 SIEMPRE — 3 rotaciones
   568→    // de frame completo + 3 intentos jsQR A con el presupuesto vencido).
   569→    for (const grados of QR_ROTACIONES) {
   570→      if (vencido()) break;
   571→      const payload = intentoSobrePixeles(rotarPixeles(pix, grados), BANDA_A);
   572→      if (payload != null) {
   573→        trazaQr(`rotacion${grados}`, payload, t0);
   574→        return payload;
   575→      }
   576→    }
   577→    logger.info(`[e14] qr-capa0 tier=null ms=${Date.now() - t0}`);
   578→    return null;
   579→  } finally {
   580→    activarGuardBarrido(null); // R1.1 — el guard muere con la cadena
   581→  }
   582→}

   730→    // lab: 3 páginas del escáner no leídas NI a 1600/2400/4000 px full-frame

   732→    if (opciones.zoomTl !== false) {
   733→      const zoomPayload = await decodificarQrZoomZbarDeCapture(dataUrl);
   734→      if (zoomPayload != null) return { payload: zoomPayload };
   735→    }
   736→    // KIT e14-gama-adaptativa-v1 — rescate 4096 OMITIDO en gama BAJO (64 MB
   737→    // puntual donde los abortos por memoria dominan — D74). Con flag OFF/
   738→    // ALTO/MEDIO rescateOn=true ⇒ byte-idéntico a hoy.
   739→    if (!rescateOn) {
   740→      logger.info(`[e14] qr-capa0 rescate4096 OMITIDO (gama adaptativa)`);
   741→      return null;
   742→    }
   743→    // R1.1 — guard de reló del rescate: vencido el presupuesto de la cadena,
   744→    // el re-decode a 16 MP + barrido a 4096 (≈4× el costo del 2048) ya no
   745→    // entra — el null honesto sube (la identidad la resolverá el OCR/capa 2).
   746→    if (vencidoCapture()) {
   747→      logger.info(
   748→        `[e14] qr-capa0 rescate4096 OMITIDO (presupuesto=${presupuestoCapture}ms vencido — R1.1)`,
   749→      );
   750→      return null;
   751→    }
   752→    // e64-qr T-1.4 — RESCATE a resolución mayor (ADAPTACIÓN DECLARADA: el
   753→    // snippet del kit usa `>=` con comentario «ya era ≥4096», internamente
   754→    // inconsistente — pix YA viene clampeado a 2048, así que `>= QR_LADO_
   755→    // LARGO` es precisamente el caso donde el rescate SÍ aporta píxeles
   756→    // REALES; se implementa el INTENT de R-4: rescate SOLO si el canvas de
   757→    // trabajo quedó clampeado (original > 2048) — sobre un original ≤2048
   758→    // el 4096 sería upscale inútil, jamás inventa detalle).
   759→    if (Math.max(pix.width, pix.height) < QR_LADO_LARGO) return null;
   760→    const tRescate = Date.now();
   761→    const pixGrande = await pixelesDeCapture(dataUrl, QR_LADO_LARGO_RESCATE);
   762→    if (!pixGrande) return null;
   763→    activarGuardBarrido(vencidoCapture); // R1.1 — el barrido 4096 también responde al reló
   764→    try {
   765→      const base = ladderBaseDePixeles(pixGrande);
   766→      if (base != null) {
   767→        trazaQr("rescate4096", base, tRescate);
   768→        return { payload: base };
   769→      }
   770→      const barrido = barridoQrDePixeles(pixGrande);
   771→      if (barrido != null) {
   772→        trazaQr("rescate4096-barrido", barrido, tRescate);
   773→        return { payload: barrido };
   774→      }
   775→      return null;
   776→    } finally {
   777→      activarGuardBarrido(null); // R1.1
   778→    }
   779→  } catch {
   780→    return null; // fallback limpio — la capa 0 jamás tumba el pipeline
   781→  }
   782→}

   784→// ─── Espera acotada del caso «original en vuelo» (kit e63m §1.2.4) ──────────
   785→/**
   786→ * Espera APLAZADA: re-chequeo SOLO-LECTURA del estado del original cada
   787→ * QR_ESPERA_INTERVALO_MS hasta QR_ESPERA_INTENTOS (techo efectivo 15 s —
   788→ * sin timers globales; DEDUP_CINTA_CIERTA=2 intacto). Devuelve "reutiliza"
   789→ * cuando el original aterriza con desenlace final (LISTA/DUPLICADA) y
   790→ * "techo" cuando vence el techo O el original sale del vuelo por otra
   791→ * puerta (ERROR / PENDIENTE / AUSENTE — esperar sería en balde: el
   792→ * reproceso completo honesto manda). Los parámetros son inyectables SOLO
   793→ * para el QA headless (timers cortos); en producción corren los default
   794→ * del kit.
   795→ */
   796→export async function esperarDesenlaceQr(
   797→  estadoDe: () => EstadoOriginalQr,
   798→  intervaloMs: number = QR_ESPERA_INTERVALO_MS,
   799→  maxIntentos: number = QR_ESPERA_INTENTOS,
   800→): Promise<"reutiliza" | "techo"> {
   801→  for (let i = 0; i < maxIntentos; i++) {
   802→    await new Promise((r) => setTimeout(r, intervaloMs));
   803→    const estado = estadoDe();
   804→    if (estado === "LISTA" || estado === "DUPLICADA") return "reutiliza";
   805→    if (estado === "ERROR" || estado === "PENDIENTE" || estado === "AUSENTE") {
   806→      return "techo";
   807→    }
   808→    // PROCESANDO → sigue en vuelo: la espera continúa (el techo manda).
   809→  }
   810→  return "techo";
   811→}
