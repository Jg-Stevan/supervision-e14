   404→   404→   404→/** D72 — KIT e14-masiva-formato-ancho-v1: título compacto del acta para la fila
   405→   405→   405→ *  en Carga Masiva: solo código del departamento, país/municipio, lugar/puesto,

   409→   409→   409→export function tituloDeActa(acta: Acta): string | null {
   410→   410→   410→  const c = acta.clasificacion;
   411→   411→   411→  if (!c) return null; // sin clasificación NUNCA hay título (honestidad)
   412→   412→   412→  const partes: string[] = [];
   413→   413→   413→
   414→   414→   414→  // 1. Código del departamento (solo código, e.g. "88" o "11")
   415→   415→   415→  if (c.departamento?.codigo) {
   416→   416→   416→    partes.push(c.departamento.codigo.trim());
   417→   417→   417→  } else if (c.departamento?.nombre && !c.municipio?.nombre) {
   418→   418→   418→    partes.push(c.departamento.nombre.trim());
   419→   419→   419→  }
   420→   420→   420→

   422→   422→   422→  if (c.municipio?.nombre) {
   423→   423→   423→    partes.push(c.municipio.nombre.trim());
   424→   424→   424→  }
   425→   425→   425→
   426→   426→   426→  // 3. Lugar / Puesto (e.g. "CONSULADO", "FRANKFURT", "CORFERIAS")

   428→   428→   428→  if (puesto) {
   429→   429→   429→    const puestoNorm = puesto.toUpperCase();
   430→   430→   430→    const munNorm = c.municipio?.nombre?.trim().toUpperCase();
   431→   431→   431→    if (!munNorm || puestoNorm !== munNorm) {
   432→   432→   432→      partes.push(puesto);
   433→   433→   433→    }
   434→   434→   434→  }
   435→   435→   435→
   436→   436→   436→  // 4. Tipo (e.g. "TRANSMISION", "DELEGADOS")
   437→   437→   437→  if (c.tipo) {
   438→   438→   438→    partes.push(c.tipo);
   439→   439→   439→  }
   440→   440→   440→
   441→   441→   441→  // 5. Mesa (zero-pad 3, e.g. "MESA 002")
   442→   442→   442→  if (c.mesa != null) {
   443→   443→   443→    partes.push(`MESA ${String(c.mesa).padStart(3, "0")}`);
   444→   444→   444→  }
   445→   445→   445→
   446→   446→   446→  // 6. Página (e.g. "PAG 1/2")
   447→   447→   447→  if (c.pagina && c.pagina.index != null && c.pagina.total != null) {
   448→   448→   448→    partes.push(`PAG ${c.pagina.index}/${c.pagina.total}`);
   449→   449→   449→  } else if (acta.pagina && acta.pagina.index != null && acta.pagina.total != null) {
   450→   450→   450→    partes.push(`PAG ${acta.pagina.index}/${acta.pagina.total}`);
   451→   451→   451→  }
   452→   452→   452→
   453→   453→   453→  const t = partes.join(" - ").toUpperCase();
   454→   454→   454→  return t.length > 0 ? t : null;
   455→   455→   455→}
