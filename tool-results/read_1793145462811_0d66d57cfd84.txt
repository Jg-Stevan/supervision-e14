   103→function relojDeMinutos(minutos: number): string {
   104→  const m = ((minutos % 1440) + 1440) % 1440;

   107→  const sufijo = h24 >= 12 ? "PM" : "AM";

   374→/** Estado de mesa derivado de sus partes (regla única, D13). */
   375→function estadoMesaDePartes(mesa: Mesa): Mesa["estadoMesa"] {
   376→  if (
   377→    mesa.transmision.p1.alerta ||
   378→    mesa.transmision.p2.alerta
   379→  ) {
   380→    return "INCOMPLETO";
   381→  }
   382→  const partesOk =
   383→    (mesa.delegados.p1 === "ok" ? 1 : 0) +
   384→    (mesa.delegados.p2 === "ok" ? 1 : 0) +
   385→    (mesa.transmision.p1.estado === "ok" ? 1 : 0) +
   386→    (mesa.transmision.p2.estado === "ok" ? 1 : 0);
   387→  if (partesOk === 4) return "COMPLETO";
   388→  if (partesOk === 0) return "NO_INICIADO";
   389→  return "EN_PROCESO";
   390→}

   398→/** pcts derivables de las mesas (≈ % de partes ok, kit T2). */
   399→function pctsDeMesas(mesas: Mesa[]): { delegado: number; transmision: number } {
   400→  const total = mesas.length * 2;
   401→  if (total === 0) return { delegado: 0, transmision: 0 };
   402→  const delegOk = mesas.reduce(
   403→    (s, m) =>
   404→      s + (m.delegados.p1 === "ok" ? 1 : 0) + (m.delegados.p2 === "ok" ? 1 : 0),
   405→    0
   406→  );
   407→  const transOk = mesas.reduce(
   408→    (s, m) =>
   409→      s +
   410→      (m.transmision.p1.estado === "ok" ? 1 : 0) +
   411→      (m.transmision.p2.estado === "ok" ? 1 : 0),
   412→    0
   413→  );
   414→  return {
   415→    delegado: Math.round((delegOk / total) * 100),
   416→    transmision: Math.round((transOk / total) * 100),
   417→  };
   418→}

   538→/**
   539→ * KIT carga-masiva-supervision-v1 (D27) — superpone el registro RUNTIME del
   540→ * digitalizador (fotos + encabezado impreso) sobre el acta base del mock.
   541→ * Las fotos del runtime MANDAN en las páginas que traen; el resto cae al
   542→ * acta base o al sintético D18. Si la mesa no tenía acta (NO_INICIADO), se
   543→ * construye una mínima honesta desde el registro runtime.
   544→ */
   545→function fusionarActaRuntime(
   546→  base: ActaMesa | null,
   547→  runtime: import("./actas-reales/actas-reales").ActaRealLista,
   548→  puesto: Puesto,
   549→  mesa: Mesa
   550→): ActaMesa {
   551→  const horaRuntime = (() => {
   552→    const d = new Date();
   553→    return `${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")} LOCAL`;
   554→  })();
   555→  const acta: ActaMesa = base
   556→    ? {
   557→        ...base,
   558→        formularios: { ...base.formularios },
   559→        historial: [...base.historial],
   560→        fotos: { ...(base.fotos ?? {}) },
   561→      }
   562→    : {
   563→        id: `#MSV-${runtime.idEnlace.replace(/_/g, "")}-${runtime.mesa}`,
   564→        mesaId: mesa.id,
   565→        puestoId: puesto.id,
   566→        horaCarga: horaRuntime,
   567→        formularios: { TRANSMISION: [], DELEGADOS: [] },
   568→        historial: [
   569→          {
   570→            hora: horaRuntime,
   571→            titulo: "CARGA MASIVA",
   572→            detalle: "Acta recibida del digitalizador (carga masiva)",
   573→            nivel: "NEUTRAL",
   574→          },
   575→        ],
   576→        version: 1,
   577→      };
   578→
   579→  // Formularios: tamaño y estado por página — runtime con foto → VALIDA;
   580→  // sin foto → lo que diga el base, o PENDIENTE.
   581→  const formularios: Record<FormularioTipo, PaginaActa[]> = {
   582→    TRANSMISION: [...acta.formularios.TRANSMISION],
   583→    DELEGADOS: [...acta.formularios.DELEGADOS],
   584→  };
   585→  for (const formulario of ["TRANSMISION", "DELEGADOS"] as FormularioTipo[]) {
   586→    const slotRuntime = runtime.fotos[formulario];
   587→    if (!slotRuntime) continue;
   588→    const total = Math.max(
   589→      slotRuntime.length,
   590→      formularios[formulario].length,
   591→      1
   592→    );
   593→    const paginas: PaginaActa[] = [];
   594→    for (let i = 0; i < total; i++) {
   595→      paginas.push({
   596→        numero: i + 1,
   597→        estado: slotRuntime[i]
   598→          ? "VALIDA"
   599→          : (formularios[formulario][i]?.estado ?? "PENDIENTE"),
   600→      });
   601→    }
   602→    formularios[formulario] = paginas;
   603→  }
   604→  acta.formularios = formularios;
   605→
   606→  // Fotos: el runtime manda donde trae foto; el base cubre el resto.
   607→  const fotos = { ...(acta.fotos ?? {}) } as NonNullable<ActaMesa["fotos"]>;
   608→  for (const formulario of ["TRANSMISION", "DELEGADOS"] as FormularioTipo[]) {
   609→    const slotRuntime = runtime.fotos[formulario];
   610→    if (!slotRuntime) continue;
   611→    const baseSlot = fotos[formulario] ?? [];
   612→    const total = Math.max(slotRuntime.length, baseSlot.length);
   613→    const fusion: (import("../domain/acta").FotoActa | null)[] = [];
   614→    for (let i = 0; i < total; i++) {
   615→      fusion.push(slotRuntime[i] ?? baseSlot[i] ?? null);
   616→    }
   617→    fotos[formulario] = fusion;
   618→  }
   619→  acta.fotos = fotos;
   620→
   621→  // Encabezado impreso verbatim del runtime (se muestra, no se interpreta).
   622→  acta.impresion = runtime.impresion;
   623→  return acta;
   624→}

   626→function coincide(puesto: Puesto, f: FiltrosPuestos | undefined): boolean {
   627→  if (!f) return true;
   628→  if (f.pais && puesto.codigoPais !== f.pais) return false;
   629→  if (f.zona && puesto.zonaId !== f.zona) return false;
   630→  if (f.puesto && puesto.puesto !== f.puesto) return false;
   631→  if (f.estado && puesto.estadoGlobal !== f.estado) return false;
   632→  const q = f.busqueda?.trim().toLowerCase();
   633→  if (q) {
   634→    const texto = [
   635→      puesto.id,
   636→      puesto.codigoPais,
   637→      puesto.pais,
   638→      puesto.zona,
   639→      puesto.puesto,
   640→    ]
   641→      .join(" ")
   642→      .toLowerCase();
   643→    if (!texto.includes(q)) return false;
   644→  }
   645→  return true;
   646→}
