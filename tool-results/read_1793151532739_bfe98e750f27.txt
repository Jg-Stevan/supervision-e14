   197→ * 180/270 — 0 nunca se aplica). El ganador se DERIVA del anillo: la rot más

   202→interface DesenlaceRot {

   207→function anilloGanadorLeer(): DesenlaceRot[] {
   208→  try {
   209→    const raw = localStorage.getItem(CLAVE_GANADOR);
   210→    if (raw == null) return [];
   211→    const parsed: unknown = JSON.parse(raw);
   212→    if (!Array.isArray(parsed)) return [];
   213→    const anillo: DesenlaceRot[] = [];
   214→    for (const e of parsed) {
   215→      if (e && typeof e === "object") {
   216→        const r = (e as { rot?: unknown }).rot;
   217→        const t = (e as { ts?: unknown }).ts;
   218→        if ((r === 90 || r === 180 || r === 270) && typeof t === "number") {
   219→          anillo.push({ rot: r, ts: t });
   220→        }
   221→      }
   222→    }
   223→    return anillo.slice(-ANILLO_TOPE); // anillo: solo los últimos 10
   224→  } catch {
   225→    return [];
   226→  }
   227→}

   229→/** Ganador actual del anillo (más frecuente; empate ⇒ más reciente) o null.
   230→ *  La recencia se deriva de la POSICIÓN en el anillo (el último índice es el
   231→ *  desenlace más nuevo) — robusta aunque dos desenlaces caigan en el mismo
   232→ *  ms (Date.now()).
   233→ *  KIT e14-orientacion-qr-v1 — EXPORTADO ADITIVO: la brújula QR lo lee para
   234→ *  ORDENAR las esquinas del rescate (lección bench v8.3: el orden fijo por
   235→ *  forma del warp erró 4/4; el anillo es señal de campo real). */
   236→export function ganadorPersistente(): Rotacion | null {
   237→  try {
   238→    const anillo = anilloGanadorLeer();
   239→    if (anillo.length === 0) return null;
   240→    const freq = new Map<Rotacion, { n: number; ultimaPos: number }>();
   241→    anillo.forEach((d, i) => {
   242→      const f = freq.get(d.rot);
   243→      if (f) {
   244→        f.n += 1;
   245→        f.ultimaPos = i;
   246→      } else {
   247→        freq.set(d.rot, { n: 1, ultimaPos: i });
   248→      }
   249→    });
   250→    let ganador: Rotacion | null = null;
   251→    for (const [rot, f] of freq) {
   252→      if (ganador === null) {
   253→        ganador = rot;
   254→        continue;
   255→      }
   256→      const g = freq.get(ganador)!;
   257→      if (f.n > g.n || (f.n === g.n && f.ultimaPos > g.ultimaPos)) ganador = rot;
   258→    }
   259→    return ganador;
   260→  } catch {
   261→    return null;
   262→  }
   263→}

   276→  } catch {
   277→    // sin storage (modo privado/cuota): el sondeo sigue con el orden de hoy
   278→  }
   279→}

   289→/** Girar una fuente canvas/img por 90/180/270° (misma equivalencia §2.1:
   290→ *  girar el crop == girar la página y recortar la caja transformada). */
   291→function girarFuente(
   292→  src: HTMLCanvasElement | HTMLImageElement,
   293→  rot: Rotacion,
   294→): HTMLCanvasElement {
   295→  if (rot === 0) {
   296→    // Normaliza a canvas (el caller trabaja SIEMPRE con canvases).
   297→    const { w, h } = dimsDe(src);
   298→    const mismo = canvasCon(w, h); // SPRINT-A A3 — factory
   299→    mismo.getContext("2d")!.drawImage(src, 0, 0);
   300→    return mismo;
   301→  }
   302→  const { w, h } = dimsDe(src);
   303→  // SPRINT-A A3 — factory (mismo estado final: lienzo limpio del tamaño
   304→  // correcto — 180: w×h; 90/270 intercambian dimensiones — antes de girar).
   305→  const tmp = canvasCon(rot === 180 ? w : h, rot === 180 ? h : w);
   306→  const ctx = tmp.getContext("2d")!;
   307→  if (rot === 180) {
   308→    ctx.translate(w, h);
   309→    ctx.rotate(Math.PI);
   310→    ctx.drawImage(src, 0, 0);
   311→    return tmp;
   312→  }
   313→  if (rot === 90) {
   314→    ctx.translate(tmp.width, 0);
   315→    ctx.rotate(Math.PI / 2);
   316→  } else {
   317→    ctx.translate(0, tmp.height);
   318→    ctx.rotate(-Math.PI / 2);
   319→  }
   320→  ctx.drawImage(src, 0, 0);
   321→  return tmp;
   322→}

   385→  }
   386→  const scores: Partial<Record<Rotacion, number>> = {};
   387→  const anclas: Partial<Record<Rotacion, number>> = {};
   388→  const conf: Partial<Record<Rotacion, number>> = {};
   389→  // Orden de sondeo (§2.1 paso 3): 0° SIEMPRE primero (baseline del MARGEN;
   390→  // el atajo T1 de lectura-sana depende de él — INTACTO) y luego el
   391→  // desafiante probable según la forma de la tira.
   392→  // D62 T1 — orden ADAPTATIVO: con ganador persistente (aplicado=1 reciente)
   393→  // se sondea [0, ganador, resto…] (el resto conserva el orden actual minus
   394→  // ganador); sin ganador ⇒ el orden de HOY sin cambio (rama h > w respetada).
   395→  // (El orden se deriva de w/h de la FUENTE del sondeo — miniatura y original
   396→  // comparten aspecto ⇒ mismo orden en ambos caminos.)
   397→  const base: Rotacion[] = h > w ? [0, 180, 90, 270] : [0, 90, 270, 180];
   398→  const orden: Rotacion[] =
   399→    gan === null || gan === 0
   400→      ? base
   401→      : [0, gan, ...base.filter((r) => r !== 0 && r !== gan)];
   402→  let mejor: Rotacion | null = null;
   403→  // D56 §3 T4 — por qué terminó el sondeo (solo para la traza, aditivo).
   404→  // e70: + "confirmada" (criterio del dueño — valor ADITIVO, los previos
   405→  // no cambian de forma; D51).
   406→  let corte: "sana" | "presupuesto" | "completo" | "confirmada" = "completo";
   407→  for (const rot of orden) {
   408→    // §2.2 presupuesto: a mitad de probe, abortar y seguir con 0° (hoy).
   409→    // El t0 es COMPARTIDO con el camino miniatura: el techo de 12 s es el
   410→    // del probe COMPLETO (mini + posible re-corrida), no por pasada.
   411→    if (rot !== 0 && performance.now() - t0 > PROBE_BUDGET_MS) {
   412→      corte = "presupuesto";
   413→      break;
   414→    }
   415→    // 0° usa los crops DIRECTOS (rec no muta canvases — no hace falta copia).
   416→    const girados = rot === 0 ? crops : crops.map((c) => girarFuente(c, rot));
   417→    const lineas = await rapidOcrRecLote(rt, girados);
   418→    const texto = lineas.map((l) => l.texto).join("\n");
   419→    const confMedia =
   420→      lineas.length === 0
   421→        ? 0
   422→        : lineas.reduce((s, l) => s + l.conf, 0) / lineas.length;
   423→    const nAnclas = ANCLAS_LECTURA.filter(([, re]) => re.test(texto)).length;
   424→    const chars = norm(texto).replace(/ /g, "").length;
   425→    scores[rot] = Math.min(TOPE_SCORE, nAnclas * 2 + confMedia * (chars / 40));
   426→    anclas[rot] = nAnclas;
   427→    conf[rot] = confMedia;
   428→    if (rot === 0) {
   429→      // e70 — MODO CONFIRMACIÓN (P4.1): el criterio del dueño es EL gate —
   430→      // quad del WORKER + anclas >= 3 Y conf >= 0.7 a 0° ⇒ la primera
   431→      // pasada rápida CONFIRMÓ la orientación ⇒ sin challengers. En este
   432→      // modo T1 sana NO aplica (su MEDIA por chars>=150 perdonaba lecturas
   433→      // basura de actas fotografiadas de lado — el falso positivo exacto
   434→      // del skip por EXIF).
   435→      if (opciones?.confirmar) {
   436→        if (
   437→          opciones.confirmar.quadWorker &&
   438→          nAnclas >= 3 &&
   439→          confMedia >= 0.7
   440→        ) {
   441→          corte = "confirmada";
   442→          break;
   443→        }
   444→        continue;
   445→      }
   446→      // D56 §3 T1 — atajo de lectura-sana: si el texto a 0° YA se lee,
   447→      // ninguna rotación puede necesitar aplicarse y ninguna debe arriesgar
   448→      // la lectura (rotar es riesgo puro). El acta sana sale en ~4 s sin
   449→      // sondear 90/270/180; la rotada del dueño leyó anclas0=0 chars=9 →
   450→      // DEBIL → sigue a los challengers ✓. Regresión nula esperada: el
   451→      // MARGEN ya protegía parcialmente; esto lo hace explícito y barato.
   452→      if (senalDeLectura(texto).nivel !== "DEBIL") {
   453→        corte = "sana";
   454→        break;
   455→      }
   456→      continue;
   457→    }
   458→    // Guard absoluto (§2.1 paso 2): score<1.0 Y conf<0.55 → NO candidata.
   459→    const candidata = (scores[rot] ?? 0) >= 1.0 || (conf[rot] ?? 0) >= 0.55;
   460→    if (!candidata) continue;
   461→    if (mejor === null || (scores[rot] ?? 0) > (scores[mejor] ?? 0)) mejor = rot;
   462→    // Atajo (§2.1 paso 3): el desafiante ya dispara contra 0° → dejar de
   463→    // sondear (ahorra las rotaciones restantes).
   464→    if ((scores[mejor] ?? 0) >= (scores[0] ?? 0) + MARGEN_ORIENTACION) break;
   465→  }
   466→  // Liberar los canvases transitorios (§2.1 paso 4 — sin retener memoria).
   467→  crops.length = 0;
   468→  const aplicado =
   469→    mejor !== null && (scores[mejor] ?? 0) >= (scores[0] ?? 0) + MARGEN_ORIENTACION;
   470→  const rot: Rotacion = aplicado ? mejor! : 0;
   471→  return { vacio: false, w, h, rot, aplicado, scores, anclas, conf, corte, orden };
   472→}

   474→/** FALLO SEGURO del mini-probe — ¿la decisión de la miniatura alcanza
   475→ *  confianza para manda? SOLO tres desenlaces se confían (todo lo demás →
   476→ *  re-corrida a resolución original, el comportamiento de HOY):
   477→ *  · corte "sana": 0° leyó texto no-DEBIL (chars ≥ 150 o cabecera E-14 o
   478→ *    ≥2 anclas — senalDeLectura). La miniatura SUB-LEE respecto a la
   479→ *    original: si ELLA ya lee, la original lee ≥ ⇒ también "sana" ⇒
   480→ *    aplicado=0 en ambos.
   481→ *  · corte "confirmada": el gate del dueño (quadWorker + anclas ≥ 3 Y
   482→ *    conf ≥ 0.7 a 0°) — su evidencia ya es fuerte por construcción.
   483→ *  · aplicado=1 con ganadora FUERTE: ≥ PROBE_MINI_ANCLAS_MIN anclas Y
   484→ *    ≥ PROBE_MINI_CONF_MIN conf en la rotación ganadora (señal estructural
   485→ *    de rotación real — el texto de una acta rotada solo se lee girado).
   486→ *  aplicado=0 débil, corte "vacio" y "presupuesto" NUNCA se confían: la
   487→ *  miniatura sub-lee ⇒ un «no rotar» suyo podría esconder una rotación que
   488→ *  la resolución completa sí encontraría (dirección peligrosa). */
   489→function miniaturaConfiada(s: SondeoSalida): boolean {
   490→  if (s.vacio) return false;
   491→  if (s.corte === "sana" || s.corte === "confirmada") return true;
   492→  if (
   493→    s.aplicado &&
   494→    (s.anclas[s.rot] ?? 0) >= PROBE_MINI_ANCLAS_MIN &&
   495→    (s.conf[s.rot] ?? 0) >= PROBE_MINI_CONF_MIN
   496→  ) {
   497→    return true;
   498→  }
   499→  return false;
   500→}

   502→/** Miniatura JPEG (q0.92 — misma calidad del re-encode de rotarDataUrl) a
   503→ *  PROBE_MINI_LADO del lado mayor, aspecto preservado. Reúsa los helpers del
   504→ *  repo: decode vía loadImageMemo (LRU compartido — el decode que ya usaba
   505→ *  cropsOrientacion) y canvasCon (factory SPRINT-A A3); el suavizado es el
   506→ *  default del navegador — la MISMA receta de decodeImagenLru/analizarPixeles
   507→ *  (fidelidad byte-idéntica del reescalado). null ante CUALQUIER fallo → el
   508→ *  sondeo corre a resolución original (comportamiento de HOY, jamás tumba). */
   509→async function miniaturaDataUrlDe(fotoOcr: string): Promise<string | null> {
   510→  try {
   511→    const img = await loadImageMemo(fotoOcr);
   512→    const w = img.naturalWidth;
   513→    const h = img.naturalHeight;
   514→    if (!w || !h) return null;
   515→    const escala = Math.min(1, PROBE_MINI_LADO / Math.max(w, h));
   516→    const ancho = Math.max(1, Math.round(w * escala));
   517→    const alto = Math.max(1, Math.round(h * escala));
   518→    const cv = canvasCon(ancho, alto);
   519→    const ctx = cv.getContext("2d");
   520→    if (!ctx) return null;
   521→    ctx.drawImage(img, 0, 0, ancho, alto);
   522→    const url = cv.toDataURL("image/jpeg", 0.92);
   523→    cv.width = 0; // liberar el canvas transitorio (§2.1 — sin retener memoria)
   524→    cv.height = 0;
   525→    return url;
   526→  } catch {
   527→    return null;
   528→  }
   529→}

   531→/** Traza del sondeo DECISOR (formato §2.1 paso 5 — los instrumentos del dueño
   532→ *  parsean por prefijo; UNA línea por probe, los campos previos NO cambian de
   533→ *  forma). `escala=` es ADITIVO AL FINAL (D51/D56): lado mayor del canvas que
   534→ *  produjo la decisión (320 = camino miniatura; >320 = re-corrida original;
   535→ *  ≤320 = foto ya pequeña — un solo sondeo). */
   536→function trazarSondeo(s: SondeoSalida, ms: number, escala: number, gan: Rotacion | null, conExif: boolean): void {
   537→  if (s.vacio) {
   538→    logger.info(
   539→      `[e14] orientacion crop=${s.w}x${s.h} rot=0 probe={0:0.00} anclas0=0 conf0=0.00 aplicado=0 ms=${ms} cortes=vacio escala=${escala}`,
   540→    );
   541→    return;
   542→  }
   543→  const probeTxt = (Object.keys(s.scores) as unknown as Rotacion[])
   544→    .map((r) => `${r}:${(s.scores[r] ?? 0).toFixed(2)}`)
   545→    .join(",");
   546→  logger.info(
   547→    `[e14] orientacion crop=${s.w}x${s.h} rot=${s.rot} probe={${probeTxt}} anclas${s.rot}=${s.anclas[s.rot] ?? 0} conf${s.rot}=${(s.conf[s.rot] ?? 0).toFixed(2)} aplicado=${s.aplicado ? 1 : 0} ms=${ms} cortes=${s.corte} orden=${s.orden.join(",")} gan=${gan ?? "-"} exif=${conExif ? 1 : 0} escala=${escala}`,
   548→  );
   549→}

   551→function resultadoDe(s: SondeoSalida, ms: number, escala: number): ProbeOrientacion {
   552→  if (s.vacio) {
   553→    return { rot: 0, aplicado: false, scores: { 0: 0 }, anclas: { 0: 0 }, conf: { 0: 0 }, ms, orientacionProbeEscala: escala };
   554→  }
   555→  return { rot: s.rot, aplicado: s.aplicado, scores: s.scores, anclas: s.anclas, conf: s.conf, ms, orientacionProbeEscala: escala };
   556→}

   558→/** Probe de orientación (§2.1). Devuelve null SOLO si no pudo correr
   559→ *  (fallo de init/det — el pipeline sigue con 0°; sin traza, «el probe ni
   560→ *  siquiera corrió»). Con det sin cajas: traza con 0:0.00 y aplicado=0
   561→ *  (AC-4 — la traza está en el 100% de las no-SIM donde el probe corrió;
   562→ *  NUNCA se alimenta a rec un lote vacío).
   563→ *
   564→ *  KIT REPARTE-3-AGENTES-v1 (Agente A, Cambio 1) — MINI-PROBE: sondea PRIMERO
   565→ *  una miniatura a PROBE_MINI_LADO (320 px del lado mayor). Si su desenlace
   566→ *  alcanza la confianza del fallo seguro (miniaturaConfiada), ESA es la
   567→ *  decisión (ahorro ~5-13 s/acta). Si NO la alcanza (o la miniatura no pudo
   568→ *  construirse/correr), se corre el sondeo a resolución original — el
   569→ *  comportamiento de HOY byte a byte — y ESA es la decisión. El presupuesto
   570→ *  de 12 s cubre TODO el probe (t0 compartido por ambas pasadas). */
   571→export async function probeOrientacion(
   572→  fotoOcr: string,
   573→  opciones?: OpcionesProbeOrientacion,
   574→): Promise<ProbeOrientacion | null> {
   575→  const t0 = performance.now();
   576→  try {
   577→    const rt = await rapidOcrInit(); // MISMO singleton (§2.2 — sin sesiones nuevas)
   578→    // D62 T1 — el ganador se lee UNA vez: miniatura y original sondean con el
   579→    // MISMO orden (mismo aspecto ⇒ mismo orden de todas formas).
   580→    const gan = ganadorPersistente();
   581→    // ¿La foto ya es una «miniatura» (lado mayor ≤ PROBE_MINI_LADO)? Escalarla
   582→    // no cambia nada: UN solo sondeo sobre la fuente real — byte-idéntico a
   583→    // HOY (el det a cap 1280 no la reduce).
   584→    const base = await loadImageMemo(fotoOcr);
   585→    const ladoMayor = Math.max(base.naturalWidth || 0, base.naturalHeight || 0);
   586→    if (ladoMayor > 0 && ladoMayor <= PROBE_MINI_LADO) {
   587→      const s = await sondeoDe(rt, fotoOcr, opciones, gan, t0);
   588→      const ms = Math.round(performance.now() - t0);
   589→      trazarSondeo(s, ms, Math.max(s.w, s.h), gan, !!opciones?.confirmar);
   590→      if (s.aplicado) anilloGanadorRegistrar(s.rot); // D62 T1.4
   591→      return resultadoDe(s, ms, Math.max(s.w, s.h));
   592→    }
   593→    // Camino MINIATURA (cualquier fallo aquí → null → camino original).
   594→    let mini: SondeoSalida | null = null;
   595→    const miniUrl = await miniaturaDataUrlDe(fotoOcr);
   596→    if (miniUrl) {
   597→      try {
   598→        mini = await sondeoDe(rt, miniUrl, opciones, gan, t0);
   599→      } catch {
   600→        mini = null; // det/rec de la miniatura falló → resolución original
   601→      }
   602→    }
   603→    if (mini && miniaturaConfiada(mini)) {
   604→      const ms = Math.round(performance.now() - t0);
   605→      trazarSondeo(mini, ms, PROBE_MINI_LADO, gan, !!opciones?.confirmar);
   606→      if (mini.aplicado) anilloGanadorRegistrar(mini.rot); // D62 T1.4
   607→      return resultadoDe(mini, ms, PROBE_MINI_LADO);
   608→    }
   609→    // FALLO SEGURO — la miniatura no alcanzó confianza (o no corrió): el
   610→    // sondeo a resolución original de SIEMPRE decide (comportamiento de HOY).
   611→    const full = await sondeoDe(rt, fotoOcr, opciones, gan, t0);
   612→    const ms = Math.round(performance.now() - t0);
   613→    trazarSondeo(full, ms, Math.max(full.w, full.h), gan, !!opciones?.confirmar);
   614→    if (full.aplicado) anilloGanadorRegistrar(full.rot); // D62 T1.4
   615→    // WAVE2 A1 — sub-decisión de la miniatura descartada por el fallo seguro
   616→    // (datos para calibración wave 3 — jamás cambia la decisión de HOY).
   617→    const res = resultadoDe(full, ms, Math.max(full.w, full.h));
   618→    if (mini) {
   619→      res.orientacionProbeMini = {
   620→        rot: mini.rot,
   621→        aplicado: mini.aplicado,
   622→        corte: mini.corte,
   623→        conf: mini.conf[mini.rot] ?? null,
   624→        anclas: mini.anclas[mini.rot] ?? null,
   625→      };
   626→    }
   627→    return res;
   628→  } catch {
   629→    return null; // §5: el probe jamás tumba el pipeline (comportamiento de hoy)
   630→  }
   631→}
