    98→/** Identificador del acta: el código de barras REAL agrupado por su
    99→ * gramática impresa (7100 | KIT(4) | TIPO(1) | VER(2) | PÁG(2) | TOTAL(2)).
   100→ * Sin 15 dígitos → el crudo tal cual; vacío → null (la caja no se pinta). */
   101→function formatoIdentificador(codigo: string): string | null {
   102→  if (!codigo) return null;
   103→  if (/^\d{15}$/.test(codigo)) {
   104→    return `X  ${codigo.slice(0, 4)} · ${codigo.slice(4, 8)} · ${codigo.slice(8, 9)} · ${codigo.slice(9, 11)} · ${codigo.slice(11, 13)} · ${codigo.slice(13, 15)}  X`;
   105→  }
   106→  return `X  ${codigo}  X`;
   107→}

   109→/** Etiqueta del formulario (misma convención del adaptador D27 —
   110→ * store-e14.ts): TRANSMISION → «TRANSMISIÓN», DELEGADOS → «CÓNSUL/
   111→ * EMBAJADOR». Sin clasificación → el tipo crudo del acta o «—». */
   112→function etiquetaFormularioDe(acta: Acta): string {
   113→  const t = acta.clasificacion?.tipo ?? acta.tipo;
   114→  if (t === "TRANSMISION") return "TRANSMISIÓN";
   115→  if (t === "DELEGADOS") return "CÓNSUL/EMBAJADOR";
   116→  return t || "—";
   117→}

   119→/** Etiqueta ORIGEN del panel (tipo de copia del acta). */
   120→function origenDe(acta: Acta): string {

   122→  if (t === "TRANSMISION") return "Transmisión";
   123→  if (t === "DELEGADOS") return "Delegados";
   124→  return t || "—";
   125→}

   127→export function ModalRevisor({

   145→  onNavegar?: (direccion: -1 | 1) => void;

   165→    expandidoRef.current = expandido;

   170→  // Escape: pantalla completa sale primero; fuera de ella cierra el modal.

   174→      if (e.key !== "Escape") return;
   175→      if (expandidoRef.current) {
   176→        setExpandido(false);
   177→        return;
   178→      }

   180→    };

   183→  }, [acta, onCerrar]);
   184→
   185→  // ── Editor (estado local del modal — el acta vive en useE14Store) ──────
   186→  const foto = acta?.fotoProcesada ?? acta?.fotoOriginal ?? null;
   187→  const editable = Boolean(acta && !acta.enviadoAutomaticamente && foto);
   188→  const [editandoRecorte, setEditandoRecorte] = useState(false);
   189→  const [rect, setRect] = useState<RectPct | null>(null);

   191→  const [editada, setEditada] = useState(false);

   199→  const [errorMotivo, setErrorMotivo] = useState(false);
   200→  const [rechazoArmado, setRechazoArmado] = useState(false);
   201→  const [rechazoHecho, setRechazoHecho] = useState(false);
   202→  const timerRechazoRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   203→

   208→    setRect(null);
   209→    setOcupado(null);
   210→    setEditada(false);

   212→    setVerOriginal(false);
   213→    setOriginalUrl(null);
   214→    setCargandoOriginal(false);
   215→    setSinOriginal(false);
   216→    setRotacionVisual(0);
   217→    setContraste("NORMAL");
   218→    setRechazoArmado(false);
   219→    setRechazoHecho(false);
   220→    setErrorMotivo(false);
   221→    setObservaciones("");
   222→    if (timerRechazoRef.current) {
   223→      clearTimeout(timerRechazoRef.current);
   224→      timerRechazoRef.current = null;
   225→    }
   226→  }, [acta?.id]);
   227→
   228→  useEffect(
   229→    () => () => {
   230→      if (timerRechazoRef.current) clearTimeout(timerRechazoRef.current);
   231→    },
   232→    [],
   233→  );

   235→  /** ROTAR: actas sin envío automático → HORNEA 90° (persiste en el acta);
   236→   *  enviadas auto → rotación VISUAL del visor (ayuda de lectura). */
   237→  const rotarFoto = async () => {
   238→    if (!acta || !foto || ocupado) return;
   239→    if (!editable) {
   240→      setRotacionVisual((r) => (r + 90) % 360);
   241→      return;
   242→    }
   243→    if (!fotoPreEdicionRef.current) fotoPreEdicionRef.current = foto;
   244→    setEditandoRecorte(false);
   245→    setRect(null);
   246→    setOcupado("rotando");
   247→    try {
   248→      const r = await rotateProcessedDataUrl(foto, 90);
   249→      useE14Store.getState().editarActaActual({ fotoProcesada: r.url });
   250→      setEditada(true);
   251→    } catch {
   252→      /* horneado fallido — la foto queda como está (honesto, sin humo) */
   253→    } finally {
   254→      setOcupado(null);
   255→    }
   256→  };
   257→
   258→  /** APLICAR recorte — mapea el rect % → píxeles naturales y hornea. */
   259→  const aplicarRecorte = async () => {
   260→    if (!acta || !foto || !rect || ocupado) return;
   261→    const img = zonaRecorteRef.current?.querySelector("img");
   262→    if (!img || !img.naturalWidth || !img.naturalHeight) return;
   263→    if (rect.w < RECT_MIN_PCT || rect.h < RECT_MIN_PCT) return;
   264→    const natRect = {
   265→      x: (rect.x / 100) * img.naturalWidth,
   266→      y: (rect.y / 100) * img.naturalHeight,
   267→      w: (rect.w / 100) * img.naturalWidth,
   268→      h: (rect.h / 100) * img.naturalHeight,
   269→    };
   270→    if (!fotoPreEdicionRef.current) fotoPreEdicionRef.current = foto;
   271→    setOcupado("recortando");
   272→    try {
   273→      const r = await recortarDataUrl(foto, natRect);
   274→      useE14Store.getState().editarActaActual({ fotoProcesada: r.url });
   275→      setEditada(true);
   276→      setEditandoRecorte(false);
   277→      setRect(null);
   278→    } catch {
   279→      /* horneado fallido — la foto queda como está (honesto, sin humo) */
   280→    } finally {
   281→      setOcupado(null);
   282→    }
   283→  };
   284→
   285→  /** DESHACER — devuelve la foto del momento ANTES de la primera edición. */
   286→  const deshacerEdicion = () => {
   287→    const original = fotoPreEdicionRef.current;
   288→    if (!original || ocupado) return;
   289→    useE14Store.getState().editarActaActual({ fotoProcesada: original });
   290→    setEditada(false);
   291→    fotoPreEdicionRef.current = null;
   292→    setEditandoRecorte(false);
   293→    setRect(null);
   294→  };
   295→
   296→  /** MOSTRAR ORIGINAL — alterna procesada ↔ cruda. La cruda viene del acta
   297→   *  (fotoOriginal) o se materializa del archivo del registro on-demand. */
   298→  const alternarOriginal = async () => {
   299→    if (cargandoOriginal || !acta) return;
   300→    if (verOriginal) {
   301→      setVerOriginal(false);
   302→      return;
   303→    }
   304→    if (acta.fotoOriginal || originalUrl) {
   305→      setVerOriginal(true);
   306→      return;
   307→    }
   308→    if (!cargarOriginal) {
   309→      setSinOriginal(true);
   310→      return;
   311→    }
   312→    setCargandoOriginal(true);
   313→    const url = await cargarOriginal(acta.id);
   314→    setCargandoOriginal(false);
   315→    if (url) {
   316→      setOriginalUrl(url);
   317→      setVerOriginal(true);
   318→    } else {
   319→      setSinOriginal(true); // sin archivo en el registro — el botón se oculta
   320→    }
   321→  };
   322→
   323→  // ── Dibujo del rectángulo de recorte (puntero = mouse o dedo) ──────────
   324→  const pctDePuntero = (
   325→    e: React.PointerEvent,
   326→  ): { x: number; y: number } | null => {
   327→    const zona = zonaRecorteRef.current;
   328→    if (!zona) return null;
   329→    const r = zona.getBoundingClientRect();
   330→    if (!r.width || !r.height) return null;
   331→    return {
   332→      x: Math.min(100, Math.max(0, ((e.clientX - r.left) / r.width) * 100)),
   333→      y: Math.min(100, Math.max(0, ((e.clientY - r.top) / r.height) * 100)),
   334→    };
   335→  };
   336→  const onRecorteDown = (e: React.PointerEvent<HTMLDivElement>) => {
   337→    if (ocupado) return;
   338→    const p = pctDePuntero(e);
   339→    if (!p) return;
   340→    // Captura fail-safe (pointerIds sintéticos/automation pueden fallar —
   341→    // el gesto de recorte jamás se aborta por eso).
   342→    try {
   343→      e.currentTarget.setPointerCapture(e.pointerId);
   344→    } catch {
   345→      /* sin captura el arrastre sigue dentro del overlay — suficiente */
   346→    }
   347→    arrastreRef.current = p;
   348→    setRect({ x: p.x, y: p.y, w: 0, h: 0 });
   349→  };
   350→  const onRecorteMove = (e: React.PointerEvent<HTMLDivElement>) => {
   351→    const inicio = arrastreRef.current;
   352→    if (!inicio) return;
   353→    const p = pctDePuntero(e);
   354→    if (!p) return;
   355→    setRect({
   356→      x: Math.min(inicio.x, p.x),
   357→      y: Math.min(inicio.y, p.y),
   358→      w: Math.abs(p.x - inicio.x),
   359→      h: Math.abs(p.y - inicio.y),
   360→    });
   361→  };

   363→    arrastreRef.current = null;
   364→  };

   376→    if (!rechazoArmado) {
   377→      setRechazoArmado(true);
   378→      return;
   379→    }
   380→    setRechazoArmado(false);
   381→    const rechazo = { tipo: "SUPERVISOR", detalle: motivo } as const;
   382→    // Parche del acta ABIERTA (la persistencia la hace sincronizarAlVolver
   383→    // al cerrar/navegar — misma plomería del editor, intacta).

   385→      status: "RECHAZADA",
   386→      rechazo,
   387→    });

   389→    // «rechazo:supervisor» con el Set anti-dup de siempre).
   390→    useE14Store
   391→      .getState()
   392→      .contarMotivoNoEnvioActa({ ...acta, status: "RECHAZADA", rechazo });

   396→    timerRechazoRef.current = setTimeout(() => {
   397→      timerRechazoRef.current = null;
   398→      setRechazoHecho(false);
   399→      onNavegar?.(1);
   400→    }, 1200);
   401→  };
   402→
   403→  if (!acta) return null;
   404→  const join = joinDeActa(acta);
   405→  const pildora = pildoraEstadoActa(acta);
   406→  const ubic = acta.ubicacion;
   407→  const rectValido = Boolean(
   408→    rect && rect.w >= RECT_MIN_PCT && rect.h >= RECT_MIN_PCT,
   409→  );
   410→
   411→  // Foto visible (MOSTRAR ORIGINAL alterna a la cruda si existe).
   412→  const fotoVisible = verOriginal
   413→    ? (acta.fotoOriginal ?? originalUrl ?? foto)
   414→    : foto;
   415→  const hayOriginal =
   416→    Boolean(acta.fotoProcesada) &&
   417→    (Boolean(acta.fotoOriginal) ||
   418→      Boolean(originalUrl) ||
   419→      (!sinOriginal && Boolean(cargarOriginal)));
   420→
   421→  // ── Datos reconocidos (clasificación DIVIPOL + acta — jamás inventar) ──
   422→  const c = acta.clasificacion;
   423→  const pais =
   424→    c?.municipio != null
   425→      ? `${c.municipio.nombre} (${c.municipio.codigo})`
   426→      : (ubic?.municipio ?? "—");
   427→  const zonaTxt = c?.zona ? `Zona ${c.zona.codigo}` : (ubic?.zona ?? "—");
   428→  const puesto =
   429→    c?.puesto != null
   430→      ? `${c.puesto.codigo} - ${c.puesto.nombre}`
   431→      : (ubic?.puesto ?? "—");
   432→  const mesa =
   433→    c?.mesa != null
   434→      ? `Mesa ${String(c.mesa).padStart(3, "0")}`
   435→      : (ubic?.mesa ?? "—");
   436→  const pagina = acta.pagina ?? c?.pagina ?? null;
   437→  const kit =
   438→    c?.kit != null
   439→      ? String(c.kit)
   440→      : c?.codigoUnico?.kit != null
   441→        ? String(c.codigoUnico.kit)
   442→        : "—";
   443→  const identificador = formatoIdentificador(
   444→    c?.codigoUnico?.crudo || acta.codigoBarras || "",
   445→  );
   446→  const lugar = c?.puesto?.nombre ?? "";
   447→  const etiquetaForm = etiquetaFormularioDe(acta);
   448→  const enviadaAuto = Boolean(acta.enviadoAutomaticamente);
   449→  const rechazadaYa = acta.status === "RECHAZADA";
   450→
   451→  return (
   452→    <div
   453→      role="dialog"
   454→      aria-modal="true"
   455→      aria-label={`Revisión de acta ${acta.titulo}`}
   456→      className="fixed inset-0 z-50 flex h-dvh flex-col overflow-hidden bg-[#070b0a] lg:flex-row"
   457→    >
   458→      {/* ════════════════ CANVAS — VISOR DEL DOCUMENTO ════════════════ */}
   459→      <main className="grid-paper-bg relative flex h-[52vh] shrink-0 flex-col overflow-hidden bg-[#060a09] lg:h-auto lg:min-w-0 lg:flex-1">
   460→        {/* ── Toolbar flotante de inspección ── */}
   461→        <div className="absolute left-1/2 top-4 z-20 flex -translate-x-1/2 flex-wrap items-center justify-center gap-x-1 gap-y-1 rounded-md border border-[#21362e] bg-[#0d1613]/90 px-3 py-1.5 font-mono text-xs text-gray-300 shadow-2xl backdrop-blur">
   462→          <button
