   101→function relojDeMinutos(minutos: number): string {
   102→  const m = ((minutos % 1440) + 1440) % 1440;

   105→  const sufijo = h24 >= 12 ? "PM" : "AM";

   373→function estadoMesaDePartes(mesa: Mesa): Mesa["estadoMesa"] {
   374→  if (
   375→    mesa.transmision.p1.alerta ||
   376→    mesa.transmision.p2.alerta
   377→  ) {
   378→    return "INCOMPLETO";
   379→  }
   380→  const partesOk =
   381→    (mesa.delegados.p1 === "ok" ? 1 : 0) +
   382→    (mesa.delegados.p2 === "ok" ? 1 : 0) +
   383→    (mesa.transmision.p1.estado === "ok" ? 1 : 0) +
   384→    (mesa.transmision.p2.estado === "ok" ? 1 : 0);
   385→  if (partesOk === 4) return "COMPLETO";
   386→  if (partesOk === 0) return "NO_INICIADO";
   387→  return "EN_PROCESO";
   388→}

   396→/** pcts derivables de las mesas (≈ % de partes ok, kit T2). */
   397→function pctsDeMesas(mesas: Mesa[]): { delegado: number; transmision: number } {
   398→  const total = mesas.length * 2;
   399→  if (total === 0) return { delegado: 0, transmision: 0 };
   400→  const delegOk = mesas.reduce(
   401→    (s, m) =>
   402→      s + (m.delegados.p1 === "ok" ? 1 : 0) + (m.delegados.p2 === "ok" ? 1 : 0),
   403→    0
   404→  );
   405→  const transOk = mesas.reduce(
   406→    (s, m) =>
   407→      s +
   408→      (m.transmision.p1.estado === "ok" ? 1 : 0) +
   409→      (m.transmision.p2.estado === "ok" ? 1 : 0),
   410→    0
   411→  );
   412→  return {
   413→    delegado: Math.round((delegOk / total) * 100),
   414→    transmision: Math.round((transOk / total) * 100),
   415→  };
   416→}

   528→function refrescarPuesto(puesto: Puesto): void {
   529→  const mesas = MESAS_POR_PUESTO[puesto.id] ?? [];
   530→  const pcts = pctsDeMesas(mesas);

   536→function coincide(puesto: Puesto, f: FiltrosPuestos | undefined): boolean {
   537→  if (!f) return true;
   538→  if (f.pais && puesto.codigoPais !== f.pais) return false;
   539→  if (f.zona && puesto.zonaId !== f.zona) return false;
   540→  if (f.puesto && puesto.puesto !== f.puesto) return false;
   541→  if (f.estado && puesto.estadoGlobal !== f.estado) return false;
   542→  const q = f.busqueda?.trim().toLowerCase();
   543→  if (q) {
   544→    const texto = [
   545→      puesto.id,
   546→      puesto.codigoPais,
   547→      puesto.pais,
   548→      puesto.zona,
   549→      puesto.puesto,
   550→    ]
   551→      .join(" ")
   552→      .toLowerCase();
   553→    if (!texto.includes(q)) return false;
   554→  }
   555→  return true;
   556→}
