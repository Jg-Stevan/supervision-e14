   133→export function ModalRevisor({
   134→  acta,
   135→  publicacion,
   136→  onVerEnMonitor,
   137→  onCerrar,
   138→  posNavegacion,
   139→  onNavegar,
   140→  cargarOriginal,
   141→}: {
   142→  acta: Acta | null;
   143→  publicacion?: PublicacionActa;
   144→  onVerEnMonitor?: PropsMasivaView["onVerEnMonitor"];
   145→  onCerrar: () => void;
   146→  /** Posición 1-based del acta abierta entre las imágenes LISTAS del lote
   147→   * (null = fuera de la lista — chevrons sin destino). */
   148→  posNavegacion?: { index: number; total: number } | null;
   149→  /** Navegación entre imágenes procesadas (orquestador: sincroniza el acta
   150→   * abierta y carga la vecina — misma puerta que ABRIR de la bandeja). */
   151→  onNavegar?: (direccion: -1 | 1) => void;
   152→  /** MOSTRAR ORIGINAL — materializa el archivo ORIGINAL del registro
   153→   * (orquestador: FileReader del File en IDB). null si no hay archivo. */
   154→  cargarOriginal?: (id: string) => Promise<string | null>;
   155→}) {
   156→  // ── Visor (estado del redesign — todo local, cero acople con el store) ──
   157→  const [verOriginal, setVerOriginal] = useState(false);
   158→  const [rotacionVisual, setRotacionVisual] = useState(0);
   159→  const [contraste, setContraste] = useState<ContrasteModo>("NORMAL");
   160→  const [expandido, setExpandido] = useState(false);
   161→  const [escala, setEscala] = useState<number | null>(null);
   162→  // MOSTRAR ORIGINAL: la foto cruda puede vivir en el acta (fotoOriginal),
   163→  // en el archivo del registro (materialización on-demand) o no existir
   164→  // (registros sin archivo — degradación honesta del toggle).
   165→  const [originalUrl, setOriginalUrl] = useState<string | null>(null);
   166→  const [cargandoOriginal, setCargandoOriginal] = useState(false);
   167→  const [sinOriginal, setSinOriginal] = useState(false);
   168→  // Espejo del expandido para el listener de Escape (lectura sin re-suscribir).
   169→  const expandidoRef = useRef(false);
   170→  useEffect(() => {
   171→    expandidoRef.current = expandido;
   172→  }, [expandido]);
   173→  // Reporte de escala estable (VisorZoom llama onZoomChange en cada gesto).
   174→  const onZoomChange = useCallback((e: number) => setEscala(e), []);
   175→
   176→  // Escape: pantalla completa sale primero; fuera de ella cierra el modal.
   177→  useEffect(() => {
   178→    if (!acta) return;
   179→    const onKeyDown = (e: KeyboardEvent) => {
   180→      if (e.key !== "Escape") return;
   181→      if (expandidoRef.current) {
   182→        setExpandido(false);
   183→        return;
   184→      }
   185→      onCerrar();
   186→    };
   187→    document.addEventListener("keydown", onKeyDown);
   188→    return () => document.removeEventListener("keydown", onKeyDown);
   189→  }, [acta, onCerrar]);
   190→
   191→  // ── Editor (estado local del modal — el acta vive en useE14Store) ──────
   192→  const foto = acta?.fotoProcesada ?? acta?.fotoOriginal ?? null;
   193→  const editable = Boolean(acta && !acta.enviadoAutomaticamente && foto);
   194→  const [editandoRecorte, setEditandoRecorte] = useState(false);
   195→  const [rect, setRect] = useState<RectPct | null>(null);
   196→  const [ocupado, setOcupado] = useState<null | "rotando" | "recortando">(null);
   197→  const [editada, setEditada] = useState(false);
   198→  const fotoPreEdicionRef = useRef<string | null>(null);
   199→  const visorRef = useRef<VisorZoomHandle>(null);
   200→  const zonaRecorteRef = useRef<HTMLDivElement>(null);
   201→  const arrastreRef = useRef<{ x: number; y: number } | null>(null);
   202→
   203→  // ── Rechazo del supervisor (motivo obligatorio + armado a 2 toques) ────
   204→  const [observaciones, setObservaciones] = useState("");
   205→  const [errorMotivo, setErrorMotivo] = useState(false);
   206→  const [rechazoArmado, setRechazoArmado] = useState(false);
   207→  const [rechazoHecho, setRechazoHecho] = useState(false);
   208→  const timerRechazoRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   209→
   210→  // Reset del visor/editor/rechazo al cambiar de acta abierta (navegación
   211→  // incluida) — y del timer de avance pendiente.
   212→  useEffect(() => {
   213→    setEditandoRecorte(false);
   214→    setRect(null);
   215→    setOcupado(null);
   216→    setEditada(false);
   217→    fotoPreEdicionRef.current = null;
   218→    setVerOriginal(false);
   219→    setOriginalUrl(null);
   220→    setCargandoOriginal(false);
   221→    setSinOriginal(false);
   222→    setRotacionVisual(0);
   223→    setContraste("NORMAL");
   224→    setRechazoArmado(false);
   225→    setRechazoHecho(false);
   226→    setErrorMotivo(false);
   227→    setObservaciones("");
   228→    if (timerRechazoRef.current) {
   229→      clearTimeout(timerRechazoRef.current);
   230→      timerRechazoRef.current = null;
   231→    }
   232→  }, [acta?.id]);
   233→
   234→  useEffect(
   235→    () => () => {
   236→      if (timerRechazoRef.current) clearTimeout(timerRechazoRef.current);
   237→    },
   238→    [],
   239→  );
   240→
   241→  /** ROTAR: actas sin envío automático → HORNEA 90° (persiste en el acta);
   242→   *  enviadas auto → rotación VISUAL del visor (ayuda de lectura). */
   243→  const rotarFoto = async () => {
   244→    if (!acta || !foto || ocupado) return;
   245→    if (!editable) {
   246→      setRotacionVisual((r) => (r + 90) % 360);
   247→      return;
   248→    }
   249→    if (!fotoPreEdicionRef.current) fotoPreEdicionRef.current = foto;
   250→    setEditandoRecorte(false);
   251→    setRect(null);
   252→    setOcupado("rotando");
   253→    try {
   254→      const r = await rotateProcessedDataUrl(foto, 90);
   255→      useE14Store.getState().editarActaActual({ fotoProcesada: r.url });
   256→      setEditada(true);
   257→    } catch {
   258→      /* horneado fallido — la foto queda como está (honesto, sin humo) */
   259→    } finally {
   260→      setOcupado(null);
   261→    }
   262→  };
   263→
   264→  /** APLICAR recorte — mapea el rect % → píxeles naturales y hornea. */
   265→  const aplicarRecorte = async () => {
   266→    if (!acta || !foto || !rect || ocupado) return;
   267→    const img = zonaRecorteRef.current?.querySelector("img");
   268→    if (!img || !img.naturalWidth || !img.naturalHeight) return;
   269→    if (rect.w < RECT_MIN_PCT || rect.h < RECT_MIN_PCT) return;
   270→    const natRect = {
   271→      x: (rect.x / 100) * img.naturalWidth,
   272→      y: (rect.y / 100) * img.naturalHeight,
   273→      w: (rect.w / 100) * img.naturalWidth,
   274→      h: (rect.h / 100) * img.naturalHeight,
   275→    };
   276→    if (!fotoPreEdicionRef.current) fotoPreEdicionRef.current = foto;
   277→    setOcupado("recortando");
   278→    try {
   279→      const r = await recortarDataUrl(foto, natRect);
   280→      useE14Store.getState().editarActaActual({ fotoProcesada: r.url });
   281→      setEditada(true);
   282→      setEditandoRecorte(false);
   283→      setRect(null);
   284→    } catch {
   285→      /* horneado fallido — la foto queda como está (honesto, sin humo) */
   286→    } finally {
   287→      setOcupado(null);
   288→    }
   289→  };
   290→
   291→  /** DESHACER — devuelve la foto del momento ANTES de la primera edición. */
   292→  const deshacerEdicion = () => {
   293→    const original = fotoPreEdicionRef.current;
   294→    if (!original || ocupado) return;
   295→    useE14Store.getState().editarActaActual({ fotoProcesada: original });
   296→    setEditada(false);
   297→    fotoPreEdicionRef.current = null;
   298→    setEditandoRecorte(false);
   299→    setRect(null);
   300→  };
   301→
   302→  /** MOSTRAR ORIGINAL — alterna procesada ↔ cruda. La cruda viene del acta
   303→   *  (fotoOriginal) o se materializa del archivo del registro on-demand. */
   304→  const alternarOriginal = async () => {
   305→    if (cargandoOriginal || !acta) return;
   306→    if (verOriginal) {
   307→      setVerOriginal(false);
   308→      return;
   309→    }
   310→    if (acta.fotoOriginal || originalUrl) {
   311→      setVerOriginal(true);
   312→      return;
   313→    }
   314→    if (!cargarOriginal) {
   315→      setSinOriginal(true);
   316→      return;
   317→    }
   318→    setCargandoOriginal(true);
   319→    const url = await cargarOriginal(acta.id);
   320→    setCargandoOriginal(false);
   321→    if (url) {
   322→      setOriginalUrl(url);
   323→      setVerOriginal(true);
   324→    } else {
   325→      setSinOriginal(true); // sin archivo en el registro — el botón se oculta
   326→    }
   327→  };
   328→
   329→  // ── Dibujo del rectángulo de recorte (puntero = mouse o dedo) ──────────
   330→  const pctDePuntero = (
   331→    e: React.PointerEvent,
   332→  ): { x: number; y: number } | null => {
   333→    const zona = zonaRecorteRef.current;
   334→    if (!zona) return null;
   335→    const r = zona.getBoundingClientRect();
   336→    if (!r.width || !r.height) return null;
   337→    return {
   338→      x: Math.min(100, Math.max(0, ((e.clientX - r.left) / r.width) * 100)),
   339→      y: Math.min(100, Math.max(0, ((e.clientY - r.top) / r.height) * 100)),
   340→    };
   341→  };
   342→  const onRecorteDown = (e: React.PointerEvent<HTMLDivElement>) => {
   343→    if (ocupado) return;
   344→    const p = pctDePuntero(e);
   345→    if (!p) return;
   346→    // Captura fail-safe (pointerIds sintéticos/automation pueden fallar —
   347→    // el gesto de recorte jamás se aborta por eso).
   348→    try {
   349→      e.currentTarget.setPointerCapture(e.pointerId);
   350→    } catch {
   351→      /* sin captura el arrastre sigue dentro del overlay — suficiente */
   352→    }
   353→    arrastreRef.current = p;
   354→    setRect({ x: p.x, y: p.y, w: 0, h: 0 });
   355→  };
   356→  const onRecorteMove = (e: React.PointerEvent<HTMLDivElement>) => {
   357→    const inicio = arrastreRef.current;
   358→    if (!inicio) return;
   359→    const p = pctDePuntero(e);
   360→    if (!p) return;
   361→    setRect({
   362→      x: Math.min(inicio.x, p.x),
   363→      y: Math.min(inicio.y, p.y),
   364→      w: Math.abs(p.x - inicio.x),
   365→      h: Math.abs(p.y - inicio.y),
   366→    });
   367→  };
   368→  const onRecorteUp = () => {
   369→    arrastreRef.current = null;
   370→  };

   383→      setRechazoArmado(true);
   384→      return;
   385→    }

   392→      rechazo,
   393→    });

   395→    // «rechazo:supervisor» con el Set anti-dup de siempre).

   397→      .getState()
   398→      .contarMotivoNoEnvioActa({ ...acta, status: "RECHAZADA", rechazo });

   400→    setRechazoHecho(true);
   401→    if (timerRechazoRef.current) clearTimeout(timerRechazoRef.current);
   402→    timerRechazoRef.current = setTimeout(() => {
   403→      timerRechazoRef.current = null;
   404→      setRechazoHecho(false);
   405→      onNavegar?.(1);
   406→    }, 1200);
   407→  };
   408→
   409→  if (!acta) return null;
   410→  const join = joinDeActa(acta);
   411→  const pildora = pildoraEstadoActa(acta);
   412→  const ubic = acta.ubicacion;
   413→  const rectValido = Boolean(
   414→    rect && rect.w >= RECT_MIN_PCT && rect.h >= RECT_MIN_PCT,
   415→  );
   416→
   417→  // Foto visible (MOSTRAR ORIGINAL alterna a la cruda si existe).
   418→  const fotoVisible = verOriginal
   419→    ? (acta.fotoOriginal ?? originalUrl ?? foto)
   420→    : foto;
   421→  const hayOriginal =
   422→    Boolean(acta.fotoProcesada) &&
   423→    (Boolean(acta.fotoOriginal) ||
   424→      Boolean(originalUrl) ||
   425→      (!sinOriginal && Boolean(cargarOriginal)));
   426→
   427→  // ── Datos reconocidos (clasificación DIVIPOL + acta — jamás inventar) ──
   428→  const c = acta.clasificacion;
   429→  const pais =
   430→    c?.municipio != null
   431→      ? `${c.municipio.nombre} (${c.municipio.codigo})`
   432→      : (ubic?.municipio ?? "—");
   433→  const zonaTxt = c?.zona ? `Zona ${c.zona.codigo}` : (ubic?.zona ?? "—");
   434→  const puesto =
   435→    c?.puesto != null
   436→      ? `${c.puesto.codigo} - ${c.puesto.nombre}`
   437→      : (ubic?.puesto ?? "—");
   438→  const mesa =
   439→    c?.mesa != null
   440→      ? `Mesa ${String(c.mesa).padStart(3, "0")}`
   441→      : (ubic?.mesa ?? "—");
   442→  const pagina = acta.pagina ?? c?.pagina ?? null;
   443→  const kit =
   444→    c?.kit != null
   445→      ? String(c.kit)
   446→      : c?.codigoUnico?.kit != null
   447→        ? String(c.codigoUnico.kit)
   448→        : "—";
   449→  const lugar = c?.puesto?.nombre ?? "";
   450→  const etiquetaForm = etiquetaFormularioDe(acta);
   451→  const enviadaAuto = Boolean(acta.enviadoAutomaticamente);
   452→  const rechazadaYa = acta.status === "RECHAZADA";
   453→
   454→  return (
   455→    <div
   456→      role="dialog"
   457→      aria-modal="true"
   458→      aria-label={`Revisión de acta ${acta.titulo}`}
   459→      // Móvil navegable (feedback del dueño): el modal COMPLETO hace scroll
   460→      // (el panel del auditor queda alcanzable); en lg vuelve el layout de
   461→      // 2 columnas con scroll interno del panel.

   895→        <PublicacionChip publicacion={publicacion} />
   896→
   897→        {/* ── Banner de rechazo (estado del acta) ── */}
   898→        {acta.rechazo && (
   899→          <p className="rounded border border-red-900/60 bg-[#1b0d0e] px-3 py-2 font-mono text-[10px] uppercase leading-relaxed tracking-wider text-red-400">
   900→            {acta.rechazo.tipo} — {acta.rechazo.detalle}
   901→          </p>
   902→        )}
   903→
   904→        <div className="mt-auto space-y-6 border-t border-[#16231e] pt-4">
   905→          {/* ── OBSERVACIONES (motivo de la decisión — obligatorio) ── */}
   906→          <div>
   907→            <label
   908→              htmlFor="masiva-observaciones-input"
   909→              className="mb-2 block font-mono text-xs font-bold uppercase tracking-wider text-gray-400"

   911→              OBSERVACIONES
   912→            </label>

   914→              id="masiva-observaciones-input"
   915→              value={observaciones}
   916→              onChange={(e) => {
   917→                setObservaciones(e.target.value);
   918→                if (errorMotivo) setErrorMotivo(false);
   919→                if (rechazoArmado) setRechazoArmado(false);
   920→              }}
   921→              placeholder="Ingrese el motivo de la decisión..."
   922→              rows={4}
   923→              aria-invalid={errorMotivo || undefined}
   924→              aria-describedby={
   925→                errorMotivo ? "masiva-observaciones-error" : undefined
   926→              }
   927→              className={`w-full resize-none rounded border bg-[#080d0b] p-3 text-xs text-gray-200 placeholder-gray-600 transition-colors focus:outline-none focus:ring-1 ${

   930→                  : "border-[#16231e] focus:border-emerald-500 focus:ring-emerald-500"

   935→                id="masiva-observaciones-error"

   937→                className="mt-1 font-mono text-[10px] uppercase tracking-wider text-red-400"

   942→          </div>
   943→
   944→          {/* ── ACCIONES DEL SUPERVISOR ── */}

   947→              <ShieldCheck className="h-4 w-4 text-emerald-400" />
   948→              <span>ACCIONES DEL SUPERVISOR</span>
   949→            </div>
   950→            {rechazoHecho ? (
   951→              <p
   952→                role="status"
   953→                className="flex w-full items-center justify-center gap-2 rounded border border-red-900/60 bg-[#1b0d0e] px-4 py-2.5 font-mono text-xs font-bold uppercase tracking-wider text-red-400"
   954→              >
   955→                <Check className="h-3.5 w-3.5" strokeWidth={2.5} />
   956→                RECHAZADA — REGISTRADA
   957→              </p>
   958→            ) : (
   959→              <button
   960→                type="button"
   961→                onClick={rechazarActa}
   962→                disabled={rechazadaYa}
   963→                title={
   964→                  rechazadaYa
   965→                    ? "Esta acta ya está rechazada"
   966→                    : "Rechazar el acta y solicitar reescaneo (motivo obligatorio)"
   967→                }
   968→                className={`flex w-full items-center justify-center gap-2 rounded border px-4 py-2.5 font-bold text-xs uppercase tracking-wider transition-colors ${
   969→                  rechazoArmado
   970→                    ? "animate-pulse border-red-500 bg-red-500/25 text-red-300"
   971→                    : "border-red-900/60 bg-[#1b0d0e] text-red-400 hover:bg-[#281315] hover:text-red-300"

   973→              >
   974→                <AlertTriangle className="h-3.5 w-3.5" strokeWidth={2} />
   975→                {rechazadaYa
   976→                  ? "RECHAZADA"
   977→                  : rechazoArmado
   978→                    ? "¿CONFIRMAR RECHAZO?"
   979→                    : "RECHAZAR (REESCANEAR)"}
   980→              </button>
   981→            )}
   982→          </div>
   983→
   984→          {/* ── Nota de auditoría ── */}
   985→          <div className="flex items-start gap-2 border-t border-[#16231e]/40 pt-2">
   986→            <Info className="mt-px h-3.5 w-3.5 shrink-0 text-gray-400" />
   987→            <span className="font-mono text-[10px] uppercase tracking-wider text-gray-500">
   988→              ESTA ACCIÓN QUEDARÁ REGISTRADA EN LA AUDITORÍA.
   989→            </span>
   990→          </div>
   991→
   992→          {/* ── Cerrar (visible siempre — el redesign no tiene scrim-clic;
   993→              al fondo del panel queda SOLO este botón, pedido del dueño) ── */}
   994→          <button
   995→            type="button"
   996→            onClick={onCerrar}
   997→            className="min-h-[44px] w-full rounded border border-[#16231e] bg-[#080d0b] px-4 font-mono text-[11px] font-bold uppercase tracking-wider text-gray-300 transition hover:border-[#2a4037] hover:text-white"
   998→          >
   999→            CERRAR
  1000→          </button>
  1001→        </div>
  1002→      </aside>
  1003→    </div>
  1004→  );
  1005→}
