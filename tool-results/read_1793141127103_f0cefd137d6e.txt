    37→    37→/** Píldora de estado de una fila — paleta de estado del repo
    38→    38→ *  (ReviewView L222-225: verde/ámbar/rojo, mono uppercase). */
    39→    39→function pildoraDe(f: FilaMasiva): { label: string; clases: string } {
    40→    40→  switch (f.estado) {
    41→    41→    case "PENDIENTE":
    42→    42→      return { label: "PENDIENTE", clases: "border-brand-borderdim text-gray-400" };
    43→    43→    // KIT e14-masiva-tiempos-v1 (PR-1): gemela por hash — SOLO visual (al
    44→    44→    // procesar reutiliza el resultado de la original; PROCESAR IGUAL fuerza).
    45→    45→    case "DUPLICADA":
    46→    46→      return { label: "DUPLICADA", clases: "border-gray-500/60 text-gray-300 bg-white/5" };
    47→    47→    case "PROCESANDO":
    48→    48→      return { label: "PROCESANDO", clases: "border-brand-borderdim text-gray-200 bg-white/5" };
    49→    49→    case "ERROR":
    50→    50→      return { label: "ERROR", clases: "border-red-500/60 text-red-400 bg-red-500/10" };
    51→    51→    // e65-dedup-enviadas-v1 (T-7, D-e65-2): nació YA-ENVIADA — SOLO-VISUAL
    52→    52→    // (cero OCR, cero envío; PROCESAR IGUAL fuerza el reproceso completo).
    53→    53→    case "YA-ENVIADA":
    54→    54→      return { label: "YA ENVIADA", clases: "border-emerald-500/60 text-emerald-300 bg-emerald-500/10" };
    55→    55→    case "LISTA": {
    56→    56→      const s = f.statusActa;
    57→    57→      if (s === "ENVIADA") {
    58→    58→        // KIT e14-masiva-ux-v1 (T3.3): la fila DISTINGUE envío real vs en
    59→    59→        // cola por offline, y auto vs manual. Paleta de estado del repo.
    60→    60→        if (f.envioEnCola)
    61→    61→          return { label: "EN COLA", clases: "border-amber-400/60 text-amber-400 bg-amber-400/10" };
    62→    62→        return f.enviadaAuto
    63→    63→          ? { label: "ENVIADA AUTO", clases: "border-brand-green/60 text-brand-green bg-brand-green/10" }
    64→    64→          : { label: "ENVIADA", clases: "border-brand-green/60 text-brand-green bg-brand-green/10" };
    65→    65→      }
    66→    66→      if (s === "OPTIMA")
    67→    67→        return { label: "ÓPTIMA", clases: "border-brand-green/60 text-brand-green" };
    68→    68→      if (s === "ADVERTENCIA")
    69→    69→        return { label: "ADVERTENCIA", clases: "border-amber-400/60 text-amber-400 bg-amber-400/10" };
    70→    70→      if (s === "RECHAZADA")
    71→    71→        return { label: "RECHAZADA", clases: "border-red-500/60 text-red-400 bg-red-500/10" };
    72→    72→      if (s === "EN_REVISION_HUMANA")
    73→    73→        return { label: "REVISIÓN HUMANA", clases: "border-amber-400/60 text-amber-400" };
    74→    74→      // e66 (P2): la gemela confirmada ya NO hereda ENVIADA/OPTIMA — su
    75→    75→      // desenlace es DUPLICADA (el envío real vive SOLO en la original).
    76→    76→      // El fondo azul de la FILA lo pinta esDuplicada (kit e64v, intacto).
    77→    77→      if (s === "DUPLICADA")
    78→    78→        return { label: "DUPLICADA", clases: "border-blue-400/60 text-blue-300 bg-blue-950/40" };
    79→    79→      return { label: "LISTA", clases: "border-brand-borderdim text-gray-400" };
    80→    80→    }
    81→    81→  }
    82→    82→}

    84→    84→/** KIT e64v — UNA sola definición de «duplicada» (compartida por el color
    85→    85→ *  azul y el orden de la bandeja — kit §3): la fila heredó desenlace de
    86→    86→ *  otra vía la maquinaria reutilizarResultado — referencia al original
    87→    87→ *  (`dupOf`) PRESENTE y desenlace heredado YA CONFIRMADO (estado final
    88→    88→ *  LISTA con statusActa). Una dup APLAZADA (estado DUPLICADA, original
    89→    89→ *  en vuelo) aún no confirmó → JAMÁS azul (kit §1). Tokens sin-match o
    90→    90→ *  ausentes no llegan aquí: sin dupOf la fila es ORIGINAL por definición

    92→    92→function esDuplicada(f: FilaMasiva): boolean {

   115→   115→/** Etiqueta de vida de la fila activa: FASE + el % ÚNICO del proceso
   116→   116→ *  COMPLETO del acta (cambio pedido por el dueño: UN solo % por acta que
   117→   117→ *  NO se reinicia entre fases — antes cada fase tenía su número y el OCR
   118→   118→ *  escalonado con 2 pasadas lo hacía retroceder 95→50→95). El % viene del
   119→   119→ *  runner (`fracGlobal`, monotónico); la fase es solo contexto. */
   120→   120→function etiquetaVida(p: { etapa: EtapaVidaKit; fracGlobal?: number }): string {
   121→   121→  const pct =
   122→   122→    p.fracGlobal != null
   123→   123→      ? `${Math.min(99, Math.max(0, Math.round(p.fracGlobal * 100)))}%`
   124→   124→      : "…";
   125→   125→  switch (p.etapa) {
   126→   126→    case "preparando":
   127→   127→      return `PREPARANDO · ${pct}`;
   128→   128→    case "deteccion":
   129→   129→      return `DETECCIÓN · ${pct}`;
   130→   130→    case "verificacion":
   131→   131→      return `VERIFICANDO · ${pct}`;
   132→   132→    case "ocr":
   133→   133→      return `OCR · ${pct}`;
   134→   134→  }
   135→   135→}
   136→   136→
   137→   137→/** % ÚNICO del PROCESO COMPLETO del acta (0% a 100%) — pedido del dueño:
   138→   138→ *  un solo número por acta, sin reinicios por fase. `fracGlobal` lo computa
   139→   139→ *  el runner (monotónico, todas las fases sumadas). LISTA = 100 · ERROR = 0 ·
   140→   140→ *  fila PROCESANDO sin señal viva (zombi de recarga) = 20 como hoy. */
   141→   141→function porcentajeProcesoActa(
   142→   142→  f: FilaMasiva,
   143→   143→  esActiva: boolean,
   144→   144→  progreso: { fracGlobal?: number } | null,
   145→   145→): number {
   146→   146→  if (f.estado === "LISTA") return 100;
   147→   147→  if (f.estado === "ERROR") return 0;
   148→   148→  if (!esActiva) return f.estado === "PROCESANDO" ? 20 : 0;
   149→   149→  if (progreso?.fracGlobal != null) {
   150→   150→    return Math.min(99, Math.max(1, Math.round(progreso.fracGlobal * 100)));
   151→   151→  }
   152→   152→  return 20;
   153→   153→}
