   128→function derivarEstadoPorUmbral(

   132→  if (sembrado === "COMPLETO") return "COMPLETO";

   137→const CIERRES_COL_MIN = [9 * 60, 10 * 60, 11 * 60, 12 * 60, 13 * 60];

   139→/**
   140→ * TIEMPO transcurrido desde el cierre (pedido del dueño): la columna
   141→ * «Tiempo» es el tiempo desde el cierre de 4:00 p. m. HORA LOCAL del
   142→ * país del puesto de digitalización (regla del kit monitor-jornada-v1:
   143→ * «cierre 4:00 p. m. hora local»); > 3 hrs es crítico. Distribución del
   144→ * demo moldeada por tramos (30% / 25% / 20% / 25%) — mapea UN draw
   145→ * uniforme a un minuto concreto dentro del tramo.
   146→ */
   147→function transcurridoPorSeed(r: number): number {
   148→  if (r < 0.3) return Math.floor((r / 0.3) * 59); // < 1 hr: 0-58 min
   149→  if (r < 0.55) return 60 + Math.floor(((r - 0.3) / 0.25) * 59); // 1-2 hrs
   150→  if (r < 0.75) return 120 + Math.floor(((r - 0.55) / 0.2) * 59); // 2-3 hrs
   151→  return 180 + Math.floor(((r - 0.75) / 0.25) * 175); // > 3 hrs: 180-354
   152→}

   154→/** "19:08" → "7:08 PM" (formato del mockup). */
   155→function relojDeMinutos(minutos: number): string {
   156→  const m = ((minutos % 1440) + 1440) % 1440;
   157→  const h24 = Math.floor(m / 60);
   158→  const mm = String(m % 60).padStart(2, "0");
   159→  const sufijo = h24 >= 12 ? "PM" : "AM";
   160→  const h12 = h24 % 12 === 0 ? 12 : h24 % 12;
   161→  return `${h12}:${mm} ${sufijo}`;
   162→}

   165→ * Horarios creíbles + determinísticos por seed (el demo no conoce
   166→ * zonas horarias reales; la coherencia EXACTA solo la exige el kit para
   167→ * el ancla Roma — §4.9).
   168→ *
   169→ * Pedido del dueño (semántica de las columnas de tiempo):
   170→ * - «Hora actual» = hora del PAÍS del puesto de digitalización — queda
   171→ *   coherente por construcción: 16:00 local (cierre) + transcurrido.
   172→ * - «Tiempo» = transcurrido desde el cierre de 4:00 p. m. LOCAL, con la
   173→ *   etiqueta del tramo canónico (domain/tiempo-cierre).
   174→ *
   175→ * D43 — el horario consume EXACTAMENTE 3 draws como antes (cierre COL,
   176→ * transcurrido, jitter quemado) para NO desplazar la posición del draw
   177→ * de estado en la secuencia determinística: los estados sembrados y los
   178→ * KPIs del demo quedan idénticos.

   180→function horarioPorSeed(random: () => number): {
   181→  cierre: string;
   182→  local: string;
   183→  diferencia: string;
   184→  minutosDesdeCierre: number;
   185→} {
   186→  // 1) Cierre en hora de Colombia (16:00 local convertido al huso).
   187→  const cierre = CIERRES_COL_MIN[Math.floor(random() * CIERRES_COL_MIN.length)];
   188→  // 2) Transcurrido desde el cierre local 4:00 p. m. (0–354 min).
   189→  const transcurrido = transcurridoPorSeed(random());
   190→  // 3) Tercer draw CONSERVADO (antes: jitter de minutos) — se quema para
   191→  //    mantener la secuencia de seeds estable (estados/KPIs inalterados).
   192→  void random();
   193→  const local = CIERRE_LOCAL_MIN + transcurrido;
   194→  const tramo = tramoDeMinutos(transcurrido);
   195→  return {
   196→    cierre: relojDeMinutos(cierre),
   197→    local: relojDeMinutos(local),
   198→    diferencia: tramo.etiqueta,
   199→    minutosDesdeCierre: transcurrido,
   200→  };
   201→}

   494→/** pcts derivables de las mesas (≈ % de partes ok, kit T2). */
   495→function pctsDeMesas(mesas: Mesa[]): { delegado: number; transmision: number } {
   496→  const total = mesas.length * 2;
   497→  if (total === 0) return { delegado: 0, transmision: 0 };
   498→  const delegOk = mesas.reduce(
   499→    (s, m) =>
   500→      s + (m.delegados.p1 === "ok" ? 1 : 0) + (m.delegados.p2 === "ok" ? 1 : 0),
   501→    0
   502→  );
   503→  const transOk = mesas.reduce(
   504→    (s, m) =>
   505→      s +
   506→      (m.transmision.p1.estado === "ok" ? 1 : 0) +
   507→      (m.transmision.p2.estado === "ok" ? 1 : 0),
   508→    0
   509→  );
   510→  return {
   511→    delegado: Math.round((delegOk / total) * 100),
   512→    transmision: Math.round((transOk / total) * 100),
   513→  };
   514→}
